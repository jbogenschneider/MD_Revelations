add_namespace = rev_sov_baltic

# The player can refuse; AI Russia attacks with a 90 percent chance.
country_event = {
	id = rev_sov_baltic.3
	title = rev_sov_baltic.3.t
	desc = rev_sov_baltic.3.d
	picture = GFX_report_event_generic_military
	is_triggered_only = yes

	trigger = { rev_sov_narva_crisis_available = yes }
	immediate = { set_country_flag = rev_sov_narva_decision_presented }

	option = {
		name = rev_sov_baltic.3.a
		ai_chance = { base = 90 }
		# Recheck the essentials if the player leaves this decision open.
		trigger = {
			has_war = no
			has_country_leader = { name = "Vladimir Putin" ruling_only = yes }
			BLR = { exists = yes }
			EST = {
				exists = yes
				is_subject = no
				has_faction_template = faction_template_nato
				NOT = { is_in_faction_with = SOV }
			}
		}
		if = {
			limit = { BLR = { NOT = { is_subject_of = SOV } } }
			set_autonomy = {
				target = BLR
				autonomy_state = autonomy_puppet_state
			}
		}
		declare_war_on = { target = EST type = annex_everything }
		if = {
			limit = { has_war_with = EST }
			set_country_flag = rev_sov_narva_attack_launched
			# Do not also launch the older 2029 Baltic escalation.
			set_country_flag = rev_sov_baltic_war_fired
			custom_effect_tooltip = rev_sov_baltic.3.nato_response_tt
			hidden_effect = {
				every_other_country = {
					limit = {
						has_faction_template = faction_template_nato
						is_in_faction_with = EST
						NOT = { has_war_with = SOV }
					}
					add_to_war = {
						targeted_alliance = EST
						enemy = SOV
						hostility_reason = asked_to_join
						single_target_only = yes
					}
				}
			}
			news_event = { id = rev_sov_baltic.4 days = 1 }
		}
	}

	option = {
		name = rev_sov_baltic.3.b
		ai_chance = { base = 10 }
		set_country_flag = rev_sov_narva_attack_declined
	}
}

# Worldwide WNN report, one day after the attack is actually launched.
news_event = {

	desc = rev_sov_baltic.4.d
	picture = GFX_news_generic_soldiers
	is_triggered_only = yes
	major = yes
	trigger = { SOV = { has_country_flag = rev_sov_narva_attack_launched } }
	option = { name = rev_sov_baltic.4.a }
}

country_event = {
	id = rev_sov_baltic.1
	is_triggered_only = yes
	hidden = no
	title = rev_sov_baltic.1.t
	desc = rev_sov_baltic.1.d

	trigger = {
		tag = SOV
		date > 2029.1.5
		#NOT = { has_war = yes }
		NOT = { has_country_flag = rev_sov_baltic_war_fired }
		OR = {
			1222 = { is_controlled_by = SOV }
			NOT = { is_ai = yes }
		}
		OR = {
			USA = { has_war_with = CAN }
			USA = { has_war_with = ENG }
			USA = { has_war_with = POL }
		}
	}

	option = {
		name = rev_sov_baltic.1.a
		ai_chance = {
			base = 0
			modifier = {
				factor = 100
				1222 = { is_controlled_by = SOV }
			}
		}
		set_country_flag = rev_sov_baltic_war_fired

		if = {
			limit = {
				BLR = {
					exists = yes
					NOT = { is_subject_of = SOV }
				}
			}
			set_autonomy = {
				target = BLR
				autonomy_state = autonomy_puppet_state
			}
		}

		if = {
			limit = {
				LAT = { exists = yes NOT = { is_subject = yes } }
				NOT = { has_war_with = LAT }
			}
			create_wargoal = {
				type = annex_everything
				target = LAT
			}
		}
		if = {
			limit = {
				EST = { exists = yes NOT = { is_subject = yes } }
				NOT = { has_war_with = EST }
			}
			create_wargoal = {
				type = annex_everything
				target = EST
			}
		}
		if = {
			limit = {
				LIT = { exists = yes NOT = { is_subject = yes } }
				NOT = { has_war_with = LIT }
			}
			create_wargoal = {
				type = annex_everything
				target = LIT
			}
			declare_war_on = {
				target = LIT
				type = annex_everything
			}
		}

		if = {
			limit = {
				UKR = {
					exists = yes
					NOT = { is_subject = yes }
					NOT = { has_idea = NATO_member }
				}
			}
			UKR = { country_event = { id = rev_sov_baltic.2 days = 30 } }
		}
	}

	option = {
		name = rev_sov_baltic.1.b
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				1222 = { is_controlled_by = SOV }
			}
		}
		set_country_flag = rev_sov_baltic_war_fired
	}
}
