	id = rev_holy_monarchy_focus.1
	title = rev_holy_monarchy_focus.1.t
	desc = rev_holy_monarchy_focus.1.d
	picture = GFX_religious_meeting
	is_triggered_only = yes
	trigger = {
		is_in_faction = no
		is_subject = no
		event_target:rev_hm_league_inviter = { has_faction_template = faction_template_holy_league is_faction_leader = yes }
	}
	option = {
		name = rev_holy_monarchy_focus.1.a
		ai_chance = { base = 5 }
		if = {
			limit = {
				is_in_faction = no
				is_subject = no
				event_target:rev_hm_league_inviter = { has_faction_template = faction_template_holy_league is_faction_leader = yes }
				NOT = {
					any_country = {
						has_faction_template = faction_template_holy_league
						has_war_with = PREV
					}
				}
			}
			event_target:rev_hm_league_inviter = { add_to_faction = ROOT }
			if = {
				limit = { is_in_faction_with = event_target:rev_hm_league_inviter }
				add_stability = 0.1
				set_temp_variable = { party_index = 23 }
				set_temp_variable = { party_popularity_increase = 0.1 }
				set_temp_variable = { temp_outlook_increase = 0.1 }
				change_relative_party_popularity = yes
			}
		}
	}
	option = {
		name = rev_holy_monarchy_focus.1.b
		ai_chance = { base = 3 }
		add_stability = -0.05
		set_temp_variable = { party_index = 23 }
		set_temp_variable = { party_popularity_increase = -0.05 }
		set_temp_variable = { temp_outlook_increase = -0.05 }
		change_relative_party_popularity = yes
		event_target:rev_hm_league_inviter = { add_opinion_modifier = { target = ROOT modifier = betrayed_our_cause } }
		set_temp_variable = { temp_opinion = -5 }
		change_the_clergy_opinion = yes
	}
}

	}
