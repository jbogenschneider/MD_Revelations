# Alex takes control of Russia: January 6, 2028, for AI and human Russia.
country_event = {
	id = endtimes.15
	title = endtimes.15.t
	desc = endtimes.15.d
	is_triggered_only = yes

	trigger = {
		date > 2028.1.5
		date < 2028.1.7
		tag = SOV
		NOT = { has_country_flag = endtimes_alex_took_over }
	}

	immediate = {
		set_country_flag = endtimes_alex_took_over
		set_global_flag = endtimes_alex_took_over
		# End every current war, including wars against countries outside NATO.
		every_other_country = {
			limit = { has_war_with = SOV }
			white_peace = SOV
		}
		retire_country_leader = yes
		set_politics = {
			ruling_party = neutrality
			elections_allowed = yes
			last_election = "2028.1.6"
			election_frequency = 144 # Next election: January 2040.
		}
		clear_array = ruling_party
		clear_array = gov_coalition_array
		add_to_array = { ruling_party = 15 } # Oligarchism in MD's party system.
		set_variable = { oligarchism_leader = 1 }
		create_country_leader = {
			name = "Alex"
			picture = "gfx/leaders/SOV/alex-thebeast.dds"
			ideology = oligarchism
			traits = { neutrality_oligarchism }
		}
		startup_politics = yes
		# Clear after politics setup so inherited and setup-added spirits are removed.
		endtimes_clear_national_spirits_except_cabal_and_csto = yes
		if = {
			limit = { has_idea = CSTO_member is_in_faction = yes }
			set_faction_leader = yes
		}
		add_ideas = endtimes_cabal
		set_country_flag = endtimes_cabal_initialized
		KAZ = {
			if = {
				limit = { exists = yes }
				add_ideas = endtimes_cabal
				set_country_flag = endtimes_cabal_initialized
			}
		}
		BLR = {
			if = {
				limit = { exists = yes }
				add_ideas = endtimes_cabal
				set_country_flag = endtimes_cabal_initialized
			}
		}
		set_variable = { debt = 0 }
		set_variable = { debt_ratio = 0 }
		set_variable = { debt_rate = 0 }
		set_temp_variable = { treasury_change = 300 }
		modify_treasury_effect = yes
		ingame_update_setup = yes
		endtimes_refresh_support_floor = yes
	}

	option = {
		name = endtimes.15.a
	}
}

# France's royal restoration: five days after the French civil-war split.

	id = endtimes.11
	title = endtimes.11.t
	desc = endtimes.11.d
	is_triggered_only = yes

		tag = FRA
		has_country_flag = rev_fra_left_split_started
		NOT = { has_country_flag = endtimes_french_monarchy_restored }
	}

	immediate = {
		set_country_flag = endtimes_french_monarchy_restored
		set_global_flag = endtimes_french_monarchy_restored
		set_country_flag = endtimes_henri_heir
		# The new monarchy breaks with the Cabal and its compromised presidency.
		if = {
			limit = { has_idea = endtimes_cabal }
			remove_ideas = endtimes_cabal
		}
		retire_country_leader = yes
		set_politics = {
			ruling_party = nationalist
			elections_allowed = no
		}
		clear_array = ruling_party
		clear_array = gov_coalition_array
		add_to_array = { ruling_party = 23 } # Monarchist in MD's party system.
		# Transfer the former nationalist parties' support to the royal settlement.
		add_to_variable = { party_pop_array^23 = party_pop_array^20 }
		add_to_variable = { party_pop_array^23 = party_pop_array^21 }
		add_to_variable = { party_pop_array^23 = party_pop_array^22 }
		set_variable = { party_pop_array^20 = 0 }
		set_variable = { party_pop_array^21 = 0 }
		set_variable = { party_pop_array^22 = 0 }
		set_country_flag = party21_banned
		set_country_flag = party22_banned
		set_country_flag = party23_banned
		clr_country_flag = party24_banned
		set_variable = { Monarchist_leader = 1 }
		create_country_leader = {
			name = "Geoffroy Saint-Michel"
			picture = "gfx/leaders/FRA/GeofSanMichele.dds"
			ideology = Monarchist
			traits = {
				nationalist_Monarchist
				military_career
			}
		}
		startup_politics = yes
		endtimes_ensure_nato_membership = yes
		if = {
			limit = { ITA = { exists = yes } }
			# Le Pen's loyalists seize the southeastern border territory for Italy.
			every_owned_state = {
				limit = { OR = { state = 857 state = 63 state = 1000 } }
				ITA = { transfer_state = PREV }
			}
			if = {
				limit = { MNC = { exists = yes } }
				MNC = {
					NATO_leave = yes
					if = {
						limit = { has_idea = CSTO_member }
						remove_from_array = { global.CSTO_member = THIS }
						remove_from_tech_sharing_group = CSTO_Tech_Share
						remove_ideas = CSTO_member
					}
				}
				ITA = { annex_country = { target = MNC transfer_troops = yes } }
			}
		}
		if = {
			limit = { owns_state = 963 SWI = { exists = yes } }
			SWI = { transfer_state = 963 }
		}
		country_event = { id = rev_fra_split.2 hours = 1 }
		# Keep the later royal settlements relative to the actual French restoration.
		SPR = {
			if = { limit = { exists = yes } country_event = { id = endtimes.17 days = 2 } }
		}
		POR = {
			if = { limit = { exists = yes } country_event = { id = endtimes.17 days = 2 } }
		}
		ENG = {
			if = { limit = { exists = yes } country_event = { id = endtimes.18 days = 10 } }
		}
		IRE = {
			if = { limit = { exists = yes } country_event = { id = endtimes.18 days = 10 } }
		}
	}

	}

	}

	immediate = {

			}
		}
		ai_chance = { factor = 90 }
		if = {
			limit = { has_idea = CSTO_member }
			CSTO_leave = yes
		}
		# Save a NATO belligerent before changing Australia's alliance.
		random_other_country = {
			limit = { has_idea = NATO_member has_war_with = USA }
			save_event_target_as = endtimes_australia_nato_war_ally
		}
		endtimes_ensure_nato_membership = yes
		if = {
			limit = { NOT = { has_war_with = USA } }
			add_to_war = {
				targeted_alliance = event_target:endtimes_australia_nato_war_ally
				enemy = USA
				hostility_reason = asked_to_join
				single_target_only = yes
			}
		}
		set_country_flag = endtimes_australia_joined_nato_war
		# add_war_support clamps at zero; reset and then set exactly 80 percent.
		add_war_support = -1
		add_war_support = 0.8
	}

	option = {
		name = endtimes.33.b
		# This is the only available option when NATO and the USA are not at war.
		ai_chance = { factor = 10 }
		if = {
			limit = { has_idea = CSTO_member }
			CSTO_leave = yes
		}
		NATO_leave = yes
		leave_faction = yes
		if = {
			limit = { CHI = { exists = yes } NOT = { has_war_with = CHI } }
			declare_war_on = { target = CHI type = annex_everything }
		}
		if = {
			limit = { SOV = { exists = yes } NOT = { has_war_with = SOV } }
			declare_war_on = { target = SOV type = annex_everything }
		}
		set_country_flag = endtimes_australia_fighting_csto
		add_war_support = -1
		add_war_support = 0.8
	}
