# USA country scope. Reorganize before the declaration, never after war begins.
rev_reorganize_nato_for_canada = {
	if = {
		limit = {
			tag = USA
			NOT = { has_global_flag = rev_nato_canada_reorganized }
			NOT = { has_war_with = CAN }
			GER = { exists = yes is_subject = no }
			ENG = { exists = yes is_subject = no }
			CAN = { exists = yes is_subject = no }
		}
		# Secure the European alliance before the USA or Britain leaves it.
		GER = {
			if = {
				limit = {
					is_faction_leader = no
					OR = { rev_is_nato_faction = yes is_in_faction_with = USA }
				}
				set_faction_leader = yes
			}
		}
		set_global_flag = rev_nato_lock_ger_leader
		every_country = {
			limit = {
				exists = yes
				NOT = { tag = USA }
				OR = {
					has_idea = NATO_member
					rev_is_nato_faction = yes
					is_in_array = { global.nato_members = THIS }
				}
			}
			set_country_flag = rev_nato_preserve_during_split
		}
		rev_nato_remove_usa_membership = yes
		# Joining Britain first establishes WestNATO under British leadership.
		ENG = {
			WestNATO_join = yes
			set_faction_leader = yes
		}
		CAN = { WestNATO_join = yes }
		DEN = { WestNATO_join = yes }
		ICE = { WestNATO_join = yes }
		NOR = { WestNATO_join = yes }
		SWE = { WestNATO_join = yes }
		HOL = { WestNATO_join = yes }
		ENG = { rev_establish_nato_westnato_trust = yes }
		set_global_flag = rev_nato_canada_reorganized
		GER = { rev_restore_continental_nato = yes }
		if = {
			limit = { has_global_flag = rev_nato_continental_roster_repaired }
			log = "[GetDateText]: [Root.GetName]: NATO reorganized under Germany; Britain leads WestNATO with Canada, Denmark, Iceland, Norway, Sweden and the Netherlands."
		}
		news_event = { id = rev_nato_reorganization.1 }
	}
}

# USA scope. Clear stale NATO records even if the USA already leads another faction.
rev_nato_remove_usa_membership = {
	if = {
		limit = { is_in_faction_with = GER has_idea = NATO_member }
		NATO_leave = yes
	}
	if = {
		limit = { is_in_faction_with = GER }
		GER = { remove_from_faction = USA }
	}
	remove_from_array = { array = global.nato_members value = USA }
	remove_ideas = NATO_member
	remove_from_tech_sharing_group = NATO_Tech_Share
}

# GER scope. Restore recorded members detached from NATO or moved to the USA after the split.
rev_restore_continental_nato = {
	if = {
		limit = {
			tag = GER
			exists = yes
			is_subject = no
			NOR = {
				rev_is_westnato_faction = yes
				rev_is_meccapact_faction = yes
				has_faction_template = faction_template_csto
				has_faction_template = faction_template_holy_league
			}
			OR = { has_global_flag = rev_nato_lock_ger_leader rev_is_nato_faction = yes has_idea = NATO_member }
		}
		if = {
			limit = {
				has_global_flag = rev_nato_canada_reorganized
				is_in_faction_with = USA
				USA = { is_faction_leader = yes }
			}
			USA = { remove_from_faction = GER }
			if = { limit = { is_in_faction_with = USA } leave_faction = yes }
		}
		if = {
			limit = { is_in_faction = no has_global_flag = rev_nato_lock_ger_leader }
			create_faction_from_template = faction_template_nato
		}
		else_if = {
			limit = { NOT = { has_global_flag = rev_nato_lock_ger_leader } }
			endtimes_ensure_nato_membership = yes
		}
		if = {
			limit = { is_in_faction = yes is_faction_leader = no NOT = { is_in_faction_with = USA } }
			set_faction_leader = yes
		}
		if = {
			limit = { rev_is_nato_faction = yes is_faction_leader = yes }
			if = {
				limit = {
					has_global_flag = rev_nato_canada_reorganized
					USA = { NOT = { has_country_flag = endtimes_usa_nato_return_completed } }
				}
				USA = { rev_nato_remove_usa_membership = yes }
			}
			rev_nato_sync_existing_membership = yes
			set_temp_variable = { rev_nato_continental_members_pending = 0 }
			every_other_country = {
				limit = {
					exists = yes
					is_subject = no
					NOR = {
						tag = USA
						has_war_with = GER
						rev_is_westnato_faction = yes
						has_idea = WestNATO_member
						rev_is_meccapact_faction = yes
						has_country_flag = endtimes_meccapact_realignment_completed
					}
					OR = {
						is_in_faction = no
						is_in_faction_with = GER
						AND = {
							has_global_flag = rev_nato_canada_reorganized
							is_in_faction_with = USA
							USA = { is_faction_leader = yes }
						}
					}
					OR = {
						has_idea = NATO_member
						is_in_array = { global.nato_members = THIS }
						has_country_flag = rev_nato_preserve_during_split
					}
				}
				if = {
					limit = { is_in_faction_with = USA NOT = { is_in_faction_with = GER } }
					USA = { remove_from_faction = PREV }
					if = { limit = { is_in_faction_with = USA } leave_faction = yes }
				}
				if = { limit = { is_in_faction = no } GER = { add_to_faction = PREV } }
				if = { limit = { is_in_faction_with = GER } rev_nato_sync_existing_membership = yes }
				else = { set_temp_variable = { rev_nato_continental_members_pending = 1 } }
			}
			if = {
				limit = {
					has_global_flag = rev_nato_canada_reorganized
					check_variable = { rev_nato_continental_members_pending = 0 }
				}
				set_global_flag = rev_nato_continental_roster_repaired
				every_country = {
					limit = { has_country_flag = rev_nato_preserve_during_split }
					clr_country_flag = rev_nato_preserve_during_split
				}
				log = "[GetDateText]: Germany leads NATO and its recorded members are in the faction."
			}
		}
	}
}
