# Concrete broadcast effects: HOI4 does not accept the previous EVENT parameter block.
# Each effect has a literal event ID and is invoked with = yes.

endtimes_broadcast_news_seal_7 = {
	if = {
		limit = { GER = { exists = yes } }
		GER = { news_event = { id = endtimeseal.7 } }
	}
	else = { news_event = { id = endtimeseal.7 } }
}

endtimes_broadcast_news_10 = {
	if = {
		limit = { SAU = { exists = yes } }
		SAU = {
			log = "[GetDateText]: [This.GetName]: Broadcasting worldwide Endtimes news endtimes.10."
			news_event = { id = endtimes.10 }
		}
	}
	else = {
		log = "[GetDateText]: [This.GetName]: Broadcasting worldwide Endtimes news endtimes.10 (Saudi Arabia unavailable)."
		news_event = { id = endtimes.10 }
	}
}
