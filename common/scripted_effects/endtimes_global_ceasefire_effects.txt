# Run once from the global daily scheduler. Each peace checks an active war;
# completed conflicts, annexed countries and unrelated wars are left alone.
endtimes_apply_global_ceasefire = {
	if = {
		limit = { NOT = { has_global_flag = endtimes_global_ceasefire_occurred } }
		set_global_flag = endtimes_global_ceasefire_occurred
		if = {
			limit = { USA = { exists = yes has_war = yes } }
			USA = { set_country_flag = endtimes_usa_at_war_before_ceasefire }
		}

		# White peace can clear capitulation; retain California's pre-peace status
		# for the player's later withdrawal settlement.
		if = {
			limit = { CAL = { exists = yes has_capitulated = yes } }
			set_global_flag = endtimes_cal_capitulated_at_ceasefire
		}

		# Test before applying any peace: only an active NATO-Russia war triggers
		# the Russian government and territorial settlement.
		if = {
			limit = {
				SOV = { exists = yes }
				any_country = {
					OR = { has_faction_template = faction_template_nato has_idea = NATO_member }
					has_war_with = SOV
				}
			}
			SOV = { endtimes_russia_ceasefire_transition = yes }
		}

		# Include California even if it has not joined WestNATO or Canada capitulated.
		if = {
			limit = { USA = { exists = yes } }
			every_country = {
				limit = {
					exists = yes
					OR = {
						tag = CAN
						tag = CAL
						has_idea = WestNATO_member
						rev_is_westnato_faction = yes
					}
					has_war_with = USA
				}
				white_peace = USA
			}
		}

		# Egypt settles with Iran and the Axis unless allied with the USA.
		if = {
			limit = {
				EGY = { exists = yes NOT = { is_in_faction_with = USA } }
			}
			every_country = {
				limit = {
					exists = yes
					OR = {
						tag = PER
						is_subject_of = PER
						is_in_faction_with = PER
						has_idea = faction_resistance_axis_member
					}
					has_war_with = EGY
				}
				white_peace = EGY
			}
		}

		# End Iran's and its subjects' participation against Israel.
		if = {
			limit = { ISR = { exists = yes } }
			every_country = {
				limit = {
					exists = yes
					OR = { tag = PER is_subject_of = PER }
					has_war_with = ISR
				}
				white_peace = ISR
			}
		}

		# End any remaining Jordanian or Egyptian war with Iran.
		if = {
			limit = { PER = { exists = yes } }
			every_country = {
				limit = {
					exists = yes
					OR = { tag = JOR tag = EGY }
					has_war_with = PER
				}
				white_peace = PER
			}
		}

		# Only China's named regional opponents: never peace out North Korea.
		if = {
			limit = { CHI = { exists = yes } }
			every_country = {
				limit = {
					exists = yes
					OR = { tag = PHI tag = JAP tag = TAI tag = KOR }
					has_war_with = CHI
				}
				white_peace = CHI
			}
		}

		# Apply the ceasefire before broadcasting it, independent of player clicks.
		endtimes_broadcast_news_38 = yes
	}
}
