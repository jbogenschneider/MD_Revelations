# Internal helper, country scope. Run only after checking the destination leader.
# CSTO's original leave effect uses ROOT, so clean its research group in THIS scope.
MeccaPact_leave_previous_alliance = {
	if = { limit = { has_idea = NATO_member } NATO_leave = yes }
	if = { limit = { has_idea = WestNATO_member } WestNATO_leave = yes }
	if = {
		limit = { is_in_array = { global.CSTO_member = THIS } }
		remove_from_array = { global.CSTO_member = THIS }
	}
	remove_ideas = CSTO_member
	remove_from_tech_sharing_group = CSTO_Tech_Share
	# A faction leader must hand over leadership before leaving. This matters
	# especially for Russia leaving CSTO and Saudi Arabia leaving an older bloc.
	if = {
		limit = { is_in_faction = yes is_faction_leader = yes }
		save_event_target_as = rev_meccapact_departing_leader
		random_other_country = {
			limit = {
				is_subject = no
				is_in_faction_with = event_target:rev_meccapact_departing_leader
			}
			set_faction_leader = yes
		}
		if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
	}
	if = { limit = { is_in_faction = yes } leave_faction = yes }
}

# SAU scope. Reuse the existing pact if leadership changed.
MeccaPact_create = {
	if = {
		limit = {
			tag = SAU
			exists = yes
			is_subject = no
			NOT = { rev_is_meccapact_faction = yes }
		}
		MeccaPact_leave_previous_alliance = yes
		if = {
			limit = { is_in_faction = no }
			set_rule = { can_create_factions = yes }
			create_faction = MeccaPact
			if = {
				limit = { is_in_faction = yes is_faction_leader = yes }
				set_country_flag = rev_meccapact_generic_founder
				log = "[GetDateText]: Saudi Arabia: created MeccaPact using generic create_faction syntax."
			}
		}
	}
	if = {
		limit = { tag = SAU rev_is_meccapact_faction = yes }
		if = {
			limit = { is_faction_leader = no }
			set_faction_leader = yes
			every_other_country = {
				limit = { is_subject = no is_in_faction_with = SAU }
				MeccaPact_enroll_member = yes
			}
		}
		MeccaPact_enroll_member = yes
	}
}

# Internal helper. The caller must already belong to the actual faction.
MeccaPact_enroll_member = {
	if = {
		limit = { rev_is_meccapact_faction = yes }
		add_ideas = MeccaPact_member
		if = {
			limit = { NOT = { is_in_array = { global.meccapact_members = THIS } } }
			add_to_array = { global.meccapact_members = THIS }
		}
		add_to_tech_sharing_group = MeccaPact_Tech_Share
	}
}

# Country scope. Use Saudi Arabia as the sole destination.
MeccaPact_join = {
	if = {
		limit = { exists = yes is_subject = no }
		save_event_target_as = rev_meccapact_applicant
		SAU = { MeccaPact_create = yes }
		if = {
			limit = {
				NOT = { rev_is_meccapact_faction = yes }
				SAU = { rev_is_meccapact_faction = yes is_faction_leader = yes }
				NOT = {
					any_other_country = {
						rev_is_meccapact_faction = yes
						has_war_with = event_target:rev_meccapact_applicant
					}
				}
			}
			MeccaPact_leave_previous_alliance = yes
			SAU = {
				add_to_faction = event_target:rev_meccapact_applicant
			}
		}
		MeccaPact_enroll_member = yes
		if = {
			limit = { rev_is_meccapact_faction = yes }
			log = "[GetDateText]: [This.GetName]: MeccaPact_join confirmed actual faction membership."
		}
		else = {
			log = "[GetDateText]: [This.GetName]: MeccaPact_join FAILED; no valid leader, incompatible war or faction transfer refused."
			if = {
				limit = { NOT = { SAU = { rev_is_meccapact_faction = yes is_faction_leader = yes } } }
				log = "[GetDateText]: [This.GetName]: MeccaPact failure detail: no recognized pact leader exists."
			}
			if = {
				limit = { is_in_faction = yes }
				log = "[GetDateText]: [This.GetName]: MeccaPact failure detail: applicant remains in another faction."
			}
		}
	}
}

# Saves from before the CSTO check can contain a second pact led by Russia.
# Turkey and Iran's completed accessions identify the original bloc.
rev_repair_meccapact_split = {
	if = {
		limit = {
			has_global_flag = endtimes_saudi_meccapact_founded
			SAU = { exists = yes is_subject = no rev_is_meccapact_faction = yes }
		}
		SAU = {
			if = {
				limit = { is_faction_leader = no has_capitulated = no }
				MeccaPact_create = yes
			}
		}
		if = {
			limit = {
				has_global_flag = endtimes_join_or_war_completed
				SAU = { is_faction_leader = yes }
				DDR = { has_country_flag = endtimes_east_germany_independent }
				SOV = {
					exists = yes
					is_faction_leader = yes
					NOR = {
						is_in_faction_with = SAU
						has_faction_template = faction_template_csto
					}
				}
				OR = {
					SOV = { has_country_flag = rev_meccapact_split_repair_pending }
					AND = {
						TUR = { has_country_flag = endtimes_turkey_alex_alliance_completed is_in_faction_with = SOV }
						PER = { has_country_flag = endtimes_iran_meccapact_completed is_in_faction_with = SOV }
					}
				}
			}
			SOV = { set_country_flag = rev_meccapact_split_repair_pending }
			every_country = {
				limit = { exists = yes NOT = { tag = SOV } is_in_faction_with = SOV }
				MeccaPact_join = yes
			}
			# Dissolve the old faction through the normal departure path once Russia is alone.
			SOV = {
				if = {
					limit = { NOT = { any_other_country = { is_in_faction_with = SOV } } }
					MeccaPact_join = yes
					if = {
						limit = { rev_is_meccapact_faction = yes }
						clr_country_flag = rev_meccapact_split_repair_pending
						log = "[GetDateText]: Reunited the two MeccaPacts under Saudi Arabia."
					}
				}
			}
		}
	}
}

# Country scope. Does not eject a country from an unrelated faction.
MeccaPact_leave = {
	remove_ideas = MeccaPact_member
	if = {
		limit = { is_in_array = { global.meccapact_members = THIS } }
		remove_from_array = { global.meccapact_members = THIS }
	}
	remove_from_tech_sharing_group = MeccaPact_Tech_Share
	if = {
		limit = { rev_is_meccapact_faction = yes }
		leave_faction = yes
	}
}
