# Country scope. Restore NATO's membership systems and actual faction alliance.
endtimes_ensure_nato_membership = {
	save_event_target_as = endtimes_nato_applicant
	if = {
		limit = { is_in_faction = yes NOT = { has_idea = NATO_member } }
		leave_faction = yes
	}
	if = {
		limit = {
			NOT = { AND = { has_idea = NATO_member is_faction_leader = yes } }
		}
		random_other_country = {
			limit = { has_idea = NATO_member is_faction_leader = yes }
			if = {
				limit = { NOT = { is_in_faction_with = event_target:endtimes_nato_applicant } }
				event_target:endtimes_nato_applicant = { leave_faction = yes }
				add_to_faction = event_target:endtimes_nato_applicant
			}
		}
	}
	if = {
		limit = { NOT = { is_in_faction = yes } }
		create_faction_from_template = faction_template_nato
	}
	if = {
		limit = { NOT = { has_idea = NATO_member } }
		for_each_scope_loop = {
			array = global.nato_members
			if = {
				limit = { exists = yes has_idea = NATO_member }
				add_opinion_modifier = { target = PREV modifier = NATO_member_modifier }
				reverse_add_opinion_modifier = { target = PREV modifier = NATO_member_modifier }
				diplomatic_relation = { country = PREV relation = military_access active = yes }
				PREV = {
					diplomatic_relation = { country = PREV relation = military_access active = yes }
				}
			}
		}
		add_ideas = NATO_member
	}
	if = {
		limit = { NOT = { is_in_array = { global.nato_members = THIS } } }
		add_to_array = { global.nato_members = THIS }
	}
	add_to_tech_sharing_group = NATO_Tech_Share
	remove_ideas = Major_Non_NATO_Ally
}
