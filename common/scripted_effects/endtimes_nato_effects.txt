# Country scope. Restore NATO's membership systems and actual faction alliance.
endtimes_ensure_nato_membership = {
	if = { limit = { has_idea = MeccaPact_member } MeccaPact_leave = yes }
	if = { limit = { has_idea = WestNATO_member } WestNATO_leave = yes }
	save_event_target_as = endtimes_nato_applicant
	if = {
		# A missing membership spirit does not mean the country left NATO.
		limit = { is_in_faction = yes NOT = { rev_is_nato_faction = yes } }
		leave_faction = yes
	}
	if = {
		limit = {
			NOT = { AND = { rev_is_nato_faction = yes is_faction_leader = yes } }
		}
		random_other_country = {
			limit = { rev_is_nato_faction = yes is_faction_leader = yes }
			if = {
				limit = { NOT = { is_in_faction_with = event_target:endtimes_nato_applicant } }
				event_target:endtimes_nato_applicant = { leave_faction = yes }
				add_to_faction = event_target:endtimes_nato_applicant
			}
		}
	}
	if = {
		limit = { NOT = { is_in_faction = yes } }
		create_faction_from_template = faction_template_nato
	}
	if = {
		limit = { NOT = { has_idea = NATO_member } }
		for_each_scope_loop = {
			array = global.nato_members
			if = {
				limit = { exists = yes has_idea = NATO_member }
				add_opinion_modifier = { target = PREV modifier = NATO_member_modifier }
				reverse_add_opinion_modifier = { target = PREV modifier = NATO_member_modifier }
				diplomatic_relation = { country = PREV relation = military_access active = yes }
				PREV = {
					diplomatic_relation = { country = PREV relation = military_access active = yes }
				}
			}
		}
		add_ideas = NATO_member
	}
	if = {
		limit = { NOT = { is_in_array = { global.nato_members = THIS } } }
		add_to_array = { global.nato_members = THIS }
	}
	add_to_tech_sharing_group = NATO_Tech_Share
	remove_ideas = Major_Non_NATO_Ally
	rev_nato_clear_accession_application = yes
}

# Membership repair must never move countries or change faction leadership.
rev_nato_sync_existing_membership = {
	if = {
		limit = { rev_is_nato_faction = yes is_subject = no NOT = { original_tag = ISR } }
		if = {
			# Legacy generic factions reject custom template rules. Their Israel
			# exclusion is enforced by the existing AI and membership callbacks.
			limit = { is_faction_leader = yes has_faction_template = faction_template_nato }
			set_faction_rule = rev_nato_joining_rule_exclude_israel
		}
		if = {
			limit = { NOT = { has_idea = NATO_member } }
			add_ideas = NATO_member
			log = "[GetDateText]: [This.GetName]: Repaired missing NATO membership spirit in existing NATO faction."
		}
		if = {
			limit = { NOT = { is_in_array = { global.nato_members = THIS } } }
			add_to_array = { global.nato_members = THIS }
		}
		add_to_tech_sharing_group = NATO_Tech_Share
		remove_ideas = Major_Non_NATO_Ally
		rev_nato_clear_accession_application = yes
	}
}

# ISR scope. Catch direct/scripted or wartime entry as well as normal invitations.
# This changes faction membership, not guarantees or participation in existing wars.
rev_nato_enforce_israel_exclusion = {
	rev_nato_clear_accession_application = yes
	if = {
		limit = {
			original_tag = ISR
			rev_is_nato_faction = yes
			is_faction_leader = no
		}
		remove_ideas = NATO_member
		if = {
			limit = { is_in_array = { global.nato_members = THIS } }
			remove_from_array = { array = global.nato_members value = THIS }
		}
		remove_from_tech_sharing_group = NATO_Tech_Share
		leave_faction = yes
		log = "[GetDateText]: [This.GetName]: Removed Israel from NATO after an ineligible direct or wartime admission."
	}
}

rev_nato_clear_accession_application = {
	if = {
		limit = {
			OR = {
				has_country_flag = NATO_Aspirant
				has_country_flag = NATO_Accession_Voting
				is_in_array = { global.NATO_Accession_Candidates = THIS }
			}
		}
		clr_country_flag = NATO_Aspirant
		clr_country_flag = NATO_Accession_Voting
		if = {
			limit = { is_in_array = { global.NATO_Accession_Candidates = THIS } }
			remove_from_array = { array = global.NATO_Accession_Candidates value = THIS }
		}
		every_other_country = {
			remove_opinion_modifier = { target = PREV modifier = NATO_aspirant }
			PREV = { remove_opinion_modifier = { target = PREV modifier = NATO_aspirant } }
			clr_country_flag = NATO_Ratified_@PREV
		}
	}
}
