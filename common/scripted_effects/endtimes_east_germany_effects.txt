		ruling_party = nationalist

	}

	set_variable = { Nat_Populism_leader = 1 }

		traits = { nationalist_Nat_Populism }
	}
	startup_politics = yes
	calculate_banned_parties = yes
	update_dirty_subideology = yes
}

# DDR scope. Join Russia's CSTO without the Cabal effects or event ROOT scoping.
endtimes_east_germany_join_csto = {
	NATO_leave = yes
	leave_faction = yes
	SOV = {
		if = {
			limit = { NOT = { is_in_faction = yes } }
			create_faction_from_template = faction_template_csto
		}
		add_ideas = CSTO_member
		if = {
			limit = { NOT = { is_in_array = { global.CSTO_member = THIS } } }
			add_to_array = { global.CSTO_member = THIS }
		}
		add_to_tech_sharing_group = CSTO_Tech_Share
		set_faction_leader = yes
		add_to_faction = DDR
	}
	add_ideas = CSTO_member
	if = {
		limit = { NOT = { is_in_array = { global.CSTO_member = THIS } } }
		add_to_array = { global.CSTO_member = THIS }
	}
	add_to_tech_sharing_group = CSTO_Tech_Share
}

# GER country scope, during the July 3, 2029 settlement.
endtimes_create_independent_east_germany = {
	every_owned_state = {
		limit = {
			OR = {
				state = 539  # Mecklenburg-Vorpommern
				state = 1079 # Brandenburg
				state = 45   # Berlin
				state = 972  # Sachsen-Anhalt
				state = 44   # Sachsen
				state = 545  # Thueringen
			}
		}
		add_core_of = DDR
	}
	# release_autonomy requires a defined subject tier; independence uses release.
	release = DDR
	DDR = {
		if = {
			limit = { exists = yes }
			if = {
				limit = { is_subject = yes }
				overlord = { set_autonomy = { target = DDR autonomy_state = autonomy_free } }
			}
			set_country_flag = endtimes_east_germany_independent
			endtimes_east_germany_join_csto = yes
			set_capital = { state = 45 }
			set_variable = { gdp_per_capita = GER.gdp_per_capita }
			set_variable = { corporate_tax_rate = GER.corporate_tax_rate }
			set_variable = { population_tax_rate = GER.population_tax_rate }
			update_laws_for_new_nation = yes
			set_temp_variable = { nation_to_copy_from = GER.id }
			copy_internal_factions = yes
			configure_religious_setup = yes
			endtimes_setup_east_germany_politics = yes
			ingame_startup_nations_effect = yes
			# Snapshot Germany's researched tech and research capacity at separation,
			# including advances made since the scenario's starting history.
			inherit_technology = GER
			set_research_slots = GER.amount_research_slots
			if = {
				limit = { has_dlc = "No Step Back" }
				load_oob = DDR_endtimes_nsb
			}
			else = { load_oob = DDR_endtimes_nonnsb }
			if = {
				limit = { has_dlc = "By Blood Alone" }
				load_oob = DDR_endtimes_air_bba
			}
			else = { load_oob = DDR_endtimes_air_legacy }
			if = {
				limit = { has_dlc = "Man the Guns" }
				load_oob = DDR_endtimes_naval_mtg
			}
			else = { load_oob = DDR_endtimes_naval_legacy }
		}
	}
}
