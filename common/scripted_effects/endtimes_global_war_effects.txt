# Bring Saudi Arabia into the French civil war before FRC can call it in.
endtimes_enroll_frc_at_global_war = {
	FRC = {
		if = {
			limit = { exists = yes }
			if = {
				limit = { is_subject = yes NOT = { rev_is_meccapact_faction = yes } }
				overlord = { set_autonomy = { target = FRC autonomy_state = autonomy_free } }
			}
			if = {
				limit = { NOT = { rev_is_meccapact_faction = yes } }
				every_other_country = {
					limit = { rev_is_meccapact_faction = yes has_war_with = FRC }
					white_peace = FRC
				}
			}
			SAU = {
				if = {
					limit = {
						exists = yes
						rev_is_meccapact_faction = yes
						FRC = { has_war_with = FRA }
						NOT = { has_war_with = FRA }
						NOT = { is_in_faction_with = FRA }
					}
					add_to_war = {
						targeted_alliance = FRC
						enemy = FRA
						hostility_reason = asked_to_join
						single_target_only = yes
					}
					if = {
						limit = { has_war_with = FRA }
						log = "[GetDateText]: Saudi Arabia joined FRC's war against France before FRC entered the MeccaPact."
					}
				}
			}
			MeccaPact_join = yes
			endtimes_apply_meccapact_ultimatum_benefits = yes
		}
	}
}

# Global scope. Check every cross-bloc pair before starting a new war.
endtimes_launch_global_war = {
	set_global_flag = endtimes_global_war_announced
	if = {
		limit = {
			NOT = {
				any_country = {
					rev_is_nato_faction = yes
					any_other_country = {
						rev_is_meccapact_faction = yes
						has_war_with = PREV
						# The French civil war alone does not count as the global war.
						OR = {
							NOT = { tag = FRC }
							PREV = { NOT = { tag = FRA } }
						}
					}
				}
			}
		}
		# Check for an existing global war before Saudi Arabia enters France's war.
		endtimes_enroll_frc_at_global_war = yes
		random_country = {
			limit = { rev_is_nato_faction = yes is_faction_leader = yes }
			save_event_target_as = endtimes_global_war_nato_leader
		}
		random_country = {
			limit = { rev_is_meccapact_faction = yes is_faction_leader = yes }
			save_event_target_as = endtimes_global_war_meccapact_leader
			declare_war_on = { target = event_target:endtimes_global_war_nato_leader type = annex_everything }
			if = {
				limit = {
					tag = SAU
					event_target:endtimes_global_war_nato_leader = { tag = GER }
					has_war_with = GER
				}
				FRC = {
					if = {
						limit = {
							exists = yes
							rev_is_meccapact_faction = yes
							ITA = { exists = yes }
							NOT = { has_war_with = ITA }
							NOT = { is_in_faction_with = ITA }
						}
						declare_war_on = { target = ITA type = annex_everything }
					}
				}
				FRA = {
					if = {
						limit = { exists = yes }
						FRC = {
							if = {
								limit = {
									exists = yes
									rev_is_meccapact_faction = yes
									NOT = { has_war_with = FRA }
									NOT = { is_in_faction_with = FRA }
								}
								declare_war_on = { target = FRA type = annex_everything }
							}
						}
						SWI = {
							if = {
								limit = {
									exists = yes
									rev_is_meccapact_faction = yes
									NOT = { has_war_with = FRA }
									NOT = { is_in_faction_with = FRA }
								}
								declare_war_on = { target = FRA type = annex_everything }
							}
						}
					}
				}
				SCO = {
					if = {
						limit = {
							exists = yes
							rev_is_meccapact_faction = yes
							ENG = { exists = yes }
							NOT = { has_war_with = ENG }
							NOT = { is_in_faction_with = ENG }
						}
						declare_war_on = { target = ENG type = annex_everything }
					}
				}
				BUL = {
					if = {
						limit = {
							exists = yes
							rev_is_meccapact_faction = yes
							GRE = { exists = yes }
							NOT = { has_war_with = GRE }
							NOT = { is_in_faction_with = GRE }
						}
						declare_war_on = { target = GRE type = annex_everything }
					}
				}
				every_country = {
					limit = {
						exists = yes
						rev_is_meccapact_faction = yes
						OR = { tag = DDR tag = CZE tag = AUS }
						NOT = { has_war_with = GER }
						NOT = { is_in_faction_with = GER }
					}
					GER = {
						if = {
							limit = { PREV = { tag = DDR } }
							create_wargoal = { target = PREV type = annex_everything }
						}
						else = { create_wargoal = { target = PREV type = puppet_wargoal_focus } }
					}
					declare_war_on = { target = GER type = annex_everything }
				}
				FRC = {
					if = {
						limit = { exists = yes }
						if = {
							limit = { MNC = { exists = yes } }
							annex_country = { target = MNC }
						}
						else_if = {
							limit = { FRA = { owns_state = 946 } }
							transfer_state = 946 # Monaco, if France already annexed it.
						}
					}
				}
				TUR = {
					if = {
						limit = {
							exists = yes
							rev_is_meccapact_faction = yes
							GRE = { exists = yes }
							NOT = { has_war_with = GRE }
							NOT = { is_in_faction_with = GRE }
						}
						declare_war_on = { target = GRE type = annex_everything }
					}
				}
				SOV = {
					if = {
						limit = {
							exists = yes
							rev_is_meccapact_faction = yes
							EST = { exists = yes }
							NOT = { has_war_with = EST }
							NOT = { is_in_faction_with = EST }
						}
						declare_war_on = { target = EST type = annex_everything }
					}
				}
			}
		}
		# Enroll both factions now rather than depending on AI call-to-arms choices.
		every_country = {
			limit = {
				rev_is_nato_faction = yes
				NOT = { tag = HOL }
				NOT = { has_war_with = event_target:endtimes_global_war_meccapact_leader }
			}
			add_to_war = {
				targeted_alliance = event_target:endtimes_global_war_nato_leader
				enemy = event_target:endtimes_global_war_meccapact_leader
				hostility_reason = asked_to_join
				single_target_only = yes
			}
		}
		every_country = {
			limit = {
				rev_is_meccapact_faction = yes
				NOT = { tag = HOL }
				NOT = { has_war_with = event_target:endtimes_global_war_nato_leader }
			}
			add_to_war = {
				targeted_alliance = event_target:endtimes_global_war_meccapact_leader
				enemy = event_target:endtimes_global_war_nato_leader
				hostility_reason = asked_to_join
				single_target_only = yes
			}
		}
	}
	else = { endtimes_enroll_frc_at_global_war = yes }
	# Check the war directly so an existing Saudi-German war also qualifies.
	SAU = {
		if = {
			limit = { exists = yes NOT = { has_country_flag = endtimes_sau_army_10_deployed } }
			if = {
				limit = { NOT = { has_war_with = GER } }
				log = "[GetDateText]: Saudi Army 10 deployment skipped because Saudi Arabia is not at war with Germany."
			}
			else_if = {
				limit = { NOT = { FRC = { controls_province = 9898 } } }
				log = "[GetDateText]: Saudi Army 10 deployment skipped because FRC does not control Montpellier (9898)."
			}
			else = {
				load_oob = SAU_endtimes_army_10
				set_country_flag = endtimes_sau_army_10_deployed
				log = "[GetDateText]: Loaded the Saudi Army 10 deployment for Montpellier (9898)."
			}
		}
		if = {
			limit = {
				exists = yes
				has_war_with = GER
				# Retain the original flag so older saves cannot deploy the same units twice.
				NOT = { has_country_flag = endtimes_sau_barcelona_special_forces_deployed }
				FRC = { controls_province = 9853 }
			}
			if = {
				limit = { NOT = { has_template = "Special Operation Forces 2027 A" } }
				# Use the standard MD special forces brigade if the saved design is absent.
				division_template = {
					name = "Special Operation Forces 2027 A"
					regiments = {
						Special_Forces = { x = 0 y = 0 }
						Special_Forces = { x = 0 y = 1 }
						Special_Forces = { x = 0 y = 2 }
						Special_Forces = { x = 1 y = 0 }
						Special_Forces = { x = 1 y = 1 }
						Special_Forces = { x = 1 y = 2 }
						Arty_Bat = { x = 2 y = 0 }
						Arty_Bat = { x = 2 y = 1 }
						Arty_Bat = { x = 2 y = 2 }
					}
					support = {
						L_Recce_Comp = { x = 0 y = 0 }
						L_Engi_Comp = { x = 0 y = 1 }
						combat_service_support_company = { x = 0 y = 2 }
					}
				}
			}
			961 = { # Languedoc-Roussillon
				create_unit = {
					owner = SAU
					prioritize_location = 9853
					country_score = { base = 100 modifier = { tag = FRC add = 100 } }
					division = "division_template = \"Special Operation Forces 2027 A\" start_manpower_factor = 1.0 start_equipment_factor = 1.0"
					count = 4
				}
			}
			set_country_flag = endtimes_sau_barcelona_special_forces_deployed
			log = "[GetDateText]: Saudi special forces deployment executed for Perpignan (9853)."
		}
	}
}
