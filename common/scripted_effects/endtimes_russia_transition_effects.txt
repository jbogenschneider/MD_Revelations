# SOV country scope. Called only for a NATO-Russia war at the Global Ceasefire.
endtimes_russia_ceasefire_transition = {
	set_country_flag = endtimes_russia_international_transition
	set_global_flag = endtimes_russia_transition_completed
	set_country_flag = rev_sov_baltic_war_fired
	set_country_flag = russia_peace_decision_done_2026
	set_country_flag = rev_russia_weekly_peace_check_done
	clr_country_flag = russia_pending_election_2026
	clr_country_flag = RUSSIA_ELECTION_ACTIVE
	clr_country_flag = russia_oligarch_council_active

	# Settle Russia's existing wars before restoring borders and independence.
	every_other_country = {
		limit = { has_war_with = SOV }
		white_peace = SOV
	}

	# Record membership before faction dissolution or idea cancellation changes it.
	every_country = {
		limit = {
			OR = { has_faction_template = faction_template_csto has_idea = CSTO_member }
		}
		set_country_flag = endtimes_csto_member_at_ceasefire
		# Also settle NATO's wars against Russia's former alliance partners.
		every_other_country = {
			limit = {
				has_war_with = PREV
				OR = {
					has_faction_template = faction_template_nato
					rev_is_westnato_faction = yes
					has_idea = NATO_member
					has_idea = WestNATO_member
				}
			}
			white_peace = PREV
		}
	}
	every_country = {
		limit = {
			has_country_flag = endtimes_csto_member_at_ceasefire
			is_faction_leader = yes
			has_faction_template = faction_template_csto
		}
		dismantle_faction = yes
	}
	every_country = {
		limit = { has_country_flag = endtimes_csto_member_at_ceasefire }
		remove_ideas = CSTO_member
		remove_from_tech_sharing_group = CSTO_Tech_Share
		clr_country_flag = endtimes_csto_member_at_ceasefire
	}
	clear_array = global.CSTO_member

	# Restore the Baltic states even if annexed, then end any puppet status.
	every_state = {
		limit = { is_core_of = EST NOT = { is_owned_by = EST } }
		transfer_state_to = EST
	}
	every_state = {
		limit = { is_core_of = LAT NOT = { is_owned_by = LAT } }
		transfer_state_to = LAT
	}
	every_state = {
		limit = { is_core_of = LIT NOT = { is_owned_by = LIT } }
		transfer_state_to = LIT
	}
	EST = {
		if = { limit = { exists = yes is_subject = yes } overlord = { set_autonomy = { target = EST autonomy_state = autonomy_free } } }
		if = { limit = { exists = yes } endtimes_ensure_nato_membership = yes }
	}
	LAT = {
		if = { limit = { exists = yes is_subject = yes } overlord = { set_autonomy = { target = LAT autonomy_state = autonomy_free } } }
		if = { limit = { exists = yes } endtimes_ensure_nato_membership = yes }
	}
	LIT = {
		if = { limit = { exists = yes is_subject = yes } overlord = { set_autonomy = { target = LIT autonomy_state = autonomy_free } } }
		if = { limit = { exists = yes } endtimes_ensure_nato_membership = yes }
	}
	# Kaliningrad becomes part of Lithuania under the Great Peace settlement.
	642 = {
		transfer_state_to = LIT
		add_core_of = LIT
	}
	BLR = {
		if = {
			limit = { exists = yes is_subject = yes }
			overlord = { set_autonomy = { target = BLR autonomy_state = autonomy_free } }
		}
	}
	endtimes_restore_ukraine_full_borders = yes
	UKR = {
		if = {
			limit = { exists = yes is_subject = yes }
			overlord = { set_autonomy = { target = UKR autonomy_state = autonomy_free } }
		}
	}

	if = {
		limit = { FAR = { exists = yes is_subject_of = CHI } }
		annex_country = { target = FAR transfer_troops = yes }
		set_global_flag = endtimes_far_east_returned_at_ceasefire
	}

	# Western conservative caretaker government. Keep the remaining spirits intact:
	# do not use startup_politics, which resets coalition ideas.
	retire_country_leader = yes
	set_politics = {
		ruling_party = democratic
		elections_allowed = yes
		last_election = "2024.10.9"
		election_frequency = 48 # Next election: October 9, 2028.
	}
	clear_array = ruling_party
	clear_array = gov_coalition_array
	add_to_array = { ruling_party = 1 } # Western conservatives.
	create_country_leader = {
		name = "Dmitri Leonidovich Orlov"
		picture = "gfx/leaders/SOV/Dmitri_Leonidovich_Orlov.dds"
		ideology = conservatism
		traits = { western_conservatism }
	}
	add_ideas = endtimes_cabal
	set_country_flag = endtimes_cabal_initialized
	add_ai_strategy = { type = befriend id = GER value = 300 }
	add_ai_strategy = { type = befriend id = POL value = 300 }
	add_ai_strategy = { type = befriend id = FIN value = 300 }
}

# SOV scope. Use the existing Far East breakaway region and transfer only Russian holdings.
endtimes_release_far_east_to_china = {
	if = {
		limit = { CHI = { exists = yes } }
		every_owned_state = {
			limit = {
				OR = {
					state = 685 # Irkutsk
					state = 686 # Buryatia
					state = 687 # Khabarovsk
					state = 688 # Yakutia
					state = 689 # Chukotka
					state = 691 # Sakhalin
					state = 692 # Primorye
					state = 1068 # Jewish Autonomous Oblast
					state = 1069 # Amur
					state = 1070 # Transbaikal
					state = 1136 # Magadan
					state = 1137 # Kamchatka
				}
			}
			add_core_of = FAR
			if = {
				limit = { country_exists = FAR }
				transfer_state_to = FAR
			}
		}
		if = {
			limit = { NOT = { country_exists = FAR } any_owned_state = { is_core_of = FAR } }
			release = FAR
		}
		FAR = {
			if = {
				limit = { exists = yes NOT = { is_subject_of = CHI } }
				if = {
					limit = { is_subject = yes }
					overlord = { set_autonomy = { target = FAR autonomy_state = autonomy_free } }
				}
				CHI = { set_autonomy = { target = FAR autonomy_state = autonomy_puppet_state } }
			}
			if = {
				limit = { is_subject_of = CHI }
				set_country_flag = rev_far_east_china_puppet
				SOV = { clr_country_flag = rev_russia_far_east_reclaim_focus_unlocked }
			}
		}
	}
}

# Also runs once for saves in which the surrender already happened.
endtimes_finalize_russia_surrender = {
	UKR = {
		every_owned_state = {
			limit = {
				is_core_of = SOV
				NOT = { is_core_of = UKR }
				NOT = { is_claimed_by = UKR }
			}
			transfer_state_to = SOV
		}
	}
	every_state = {
		limit = { is_core_of = UKR is_claimed_by = SOV }
		remove_claim_by = SOV
	}
	endtimes_release_far_east_to_china = yes
	# Change only Ukrainian-controlled provinces in Russia's remaining states.
	every_owned_state = {
		limit = {
			NOT = { is_core_of = UKR }
			NOT = { is_claimed_by = UKR }
		}
		set_state_province_controller = {
			controller = SOV
			limit = { tag = UKR }
		}
	}
	GEO = {
		if = {
			limit = { exists = yes }
			if = {
				limit = { ABK = { exists = yes } }
				annex_country = { target = ABK }
			}
			else_if = {
				limit = { SOV = { owns_state = 706 } }
				transfer_state = 706 # Abkhazia, if Russia already annexed it.
			}
			every_other_country = {
				limit = { has_war_with = GEO }
				white_peace = GEO
			}
			MeccaPact_leave = yes
			if = {
				limit = { NOT = { has_government = democratic } }
				set_country_leader_ideology = conservatism
				clear_array = ruling_party
				clear_array = gov_coalition_array
				add_to_array = { ruling_party = 1 } # Western conservatism.
				clr_country_flag = party2_banned
			}
			set_politics = { ruling_party = democratic elections_allowed = yes }
			startup_politics = yes
		}
	}
	set_country_flag = endtimes_russia_surrender_aftermath_completed
	log = "[GetDateText]: Russia surrender aftermath applied, including Ukrainian territory cleanup."
}

# SOV scope. Choose the overlord before the surrender ends Russia's current wars.
endtimes_apply_russia_surrender = {
	if = {
		limit = { has_war_with = UKR }
		UKR = { save_event_target_as = endtimes_russia_surrender_overlord }
	}
	else = { POL = { save_event_target_as = endtimes_russia_surrender_overlord } }

	MeccaPact_leave_previous_alliance = yes
	MeccaPact_leave = yes
	event_target:endtimes_russia_surrender_overlord = {
		puppet = { target = SOV end_wars = yes end_civil_wars = yes }
	}
	if = {
		limit = { is_subject_of = event_target:endtimes_russia_surrender_overlord }
		set_country_flag = endtimes_russia_surrender_completed
		if = {
			limit = { POL = { exists = yes } }
			642 = {
				transfer_state_to = POL
				remove_core_of = SOV
			}
		}
		endtimes_restore_ukraine_full_borders = yes
		set_country_flag = russia_peace_decision_done_2026
		set_country_flag = rev_russia_weekly_peace_check_done
		clr_country_flag = russia_pending_election_2026
		clr_country_flag = RUSSIA_ELECTION_ACTIVE
		clr_country_flag = russia_oligarch_council_active
		retire_country_leader = yes
		set_politics = { ruling_party = democratic elections_allowed = yes }
		clear_array = ruling_party
		clear_array = gov_coalition_array
		add_to_array = { ruling_party = 2 } # Western liberals.
		clr_country_flag = party3_banned
		create_country_leader = {
			name = "Ilya Yashin"
			picture = "gfx/leaders/SOV/ilya_yashin.dds"
			ideology = liberalism
			traits = { western_liberalism }
		}
		startup_politics = yes
		ingame_update_setup = yes
		log = "[GetDateText]: Russia surrendered and accepted a new overlord."
		endtimes_finalize_russia_surrender = yes
	}
	else = { log = "[GetDateText]: Russia's scripted surrender failed to establish the new subject relationship." }
}
