# SOV country scope. Keep Orlov if he rules; otherwise install him in a coup.
endtimes_russia_suspend_october_elections = {
	if = {
		limit = {
			tag = SOV
			NOT = { has_country_flag = endtimes_russia_october_coup_completed }
			NOT = { has_country_flag = endtimes_russia_october_elections_completed }
		}
		# Record the actual leader, rather than relying on an old transition flag.
		if = {
			limit = {
				has_country_flag = endtimes_russia_international_transition
				has_country_leader = { name = "Dmitri Leonidovich Orlov" ruling_only = yes }
			}
			set_global_flag = endtimes_russia_october_coup_followed_interim
		}
		set_country_flag = endtimes_russia_october_coup_completed
		set_global_flag = endtimes_russia_october_coup_completed
		if = {
			limit = { NOT = { has_country_leader = { name = "Dmitri Leonidovich Orlov" ruling_only = yes } } }
			set_global_flag = endtimes_russia_october_coup_installed_orlov
			retire_country_leader = yes
		}
		else = { set_country_leader_ideology = Autocracy }
		set_politics = { ruling_party = communism elections_allowed = no }
		if = {
			limit = { NOT = { has_country_leader = { name = "Dmitri Leonidovich Orlov" ruling_only = yes } } }
			create_country_leader = {
				name = "Dmitri Leonidovich Orlov"
				picture = "gfx/leaders/SOV/Dmitri_Leonidovich_Orlov.dds"
				ideology = Autocracy
				traits = { career_politician }
			}
		}
		clear_array = ruling_party
		clear_array = gov_coalition_array
		add_to_array = { ruling_party = 7 } # Emerging Autocracy.
		set_popularities = {
			democratic = 29
			communism = 45
			fascism = 1
			neutrality = 10
			nationalist = 15
		}
		startup_politics = yes
		clr_country_flag = russia_pending_election_2026
		clr_country_flag = RUSSIA_ELECTION_ACTIVE

		if = {
			limit = { has_idea = orthodox_christian }
			swap_ideas = { remove_idea = orthodox_christian add_idea = christian }
		}
		else = { add_ideas = christian }

		# This coup's debt deal is 50%, including for an existing SOLOMON holder.
		# Do not call the standard 80% adoption effect or broadcast its report.
		add_ideas = Solomon_one_world_currency
		multiply_variable = { debt = 0.50 }
		set_country_flag = rev_solomon_currency_adopted
		endtimes_join_meccapact_and_cabal = yes

		# Empty the recruitable pool without deleting population or deployed units.
		set_temp_variable = { endtimes_russia_manpower_to_remove = manpower }
		multiply_temp_variable = { endtimes_russia_manpower_to_remove = -1 }
		add_manpower = endtimes_russia_manpower_to_remove
		ingame_update_setup = yes
		if = {
			limit = { NOT = { has_global_flag = endtimes_russia_coup_news_dispatched } }
			set_global_flag = endtimes_russia_coup_news_dispatched
			log = "[GetDateText]: Russia: broadcasting worldwide coup report endtimes.40."
			news_event = { id = endtimes.40 }
		}
	}
}

# SOV country scope. Elect a new Western liberal president on either prior path.
endtimes_russia_hold_october_elections = {
	set_country_flag = endtimes_russia_october_elections_completed
	clr_country_flag = russia_pending_election_2026
	clr_country_flag = RUSSIA_ELECTION_ACTIVE
	clr_country_flag = russia_oligarch_council_active
	set_temp_variable = { endtimes_russia_replace_yashin = 0 }
	if = {
		limit = { has_country_leader = { name = "Ilya Yashin" ruling_only = yes } }
		set_temp_variable = { endtimes_russia_replace_yashin = 1 }
	}
	retire_country_leader = yes
	set_politics = {
		ruling_party = democratic
		elections_allowed = yes
		last_election = "2028.10.7"
		election_frequency = 48
	}
	clear_array = ruling_party
	clear_array = gov_coalition_array
	add_to_array = { ruling_party = 2 } # Western liberals.
	if = {
		limit = { check_variable = { endtimes_russia_replace_yashin = 1 } }
		clear_array = ruling_party
		add_to_array = { ruling_party = 1 } # Western conservatives.
		create_country_leader = {
			name = "Boris Nadezhdin"
			picture = "gfx/leaders/SOV/boris_nadezhdin.dds"
			ideology = conservatism
			traits = { western_conservatism }
		}
	}
	else = {
		create_country_leader = {
			name = "Ilya Yashin"
			picture = "gfx/leaders/SOV/ilya_yashin.dds"
			ideology = liberalism
			traits = { western_liberalism }
		}
	}
	set_popularities = {
		democratic = 39
		communism = 15
		fascism = 1
		neutrality = 30
		nationalist = 15
	}
	startup_politics = yes
	set_stability = 0.80
	ingame_update_setup = yes
	news_event = { id = endtimes.43 }
}
