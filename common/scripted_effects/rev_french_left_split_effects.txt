# Country scope. Restore Western democracy and NATO membership independently.
rev_fra_split_align_nato_neighbor = {
	if = {
		limit = {
			OR = {
				NOT = { has_government = democratic }
				has_country_leader_ideology = Western_Autocracy
			}
		}
		set_country_leader_ideology = conservatism
		set_politics = { ruling_party = democratic elections_allowed = yes }
		clear_array = ruling_party
		clear_array = gov_coalition_array
		add_to_array = { ruling_party = 1 } # Western conservatism.
		clr_country_flag = party2_banned
		startup_politics = yes
	}
	if = {
		limit = { has_elections = no }
		set_politics = { elections_allowed = yes }
	}
	if = {
		limit = {
			NOT = { rev_is_nato_faction = yes }
			# Spain's temporary withdrawal lasts until NATO Recovery.
			OR = {
				NOT = { tag = SPR }
				NOT = { has_global_flag = endtimes_nato_withdrawals_completed }
				has_global_flag = endtimes_nato_recovery_completed
			}
		}
		endtimes_ensure_nato_membership = yes
	}
}

# The legacy French rosters have four regional formations. Tag Provence's fifth.
rev_fra_tag_provence_formation = {
	set_country_flag = rev_fra_provence_formation_tagged
	create_unit = {
		owner = FRA
		id = 9905801
		prioritize_location = 958
		division = "name = \"6ème Brigade légère blindée\" location = 958 division_template = \"Light Armoured Brigade\" start_experience_factor = 0.75 start_equipment_factor = 0.01"
	}
}

# FRA scope. Called by the November split event, or manually for testing.
rev_fra_start_left_split = {
	if = {
		limit = { rev_fra_left_split_ready = yes }
		every_other_country = {
			limit = { exists = yes OR = { tag = ITA tag = SPR } }
			rev_fra_split_align_nato_neighbor = yes
		}
		# Add cores only at the split, so FRC stays dormant in the 2026 scenario.
		56 = { add_core_of = FRC }
		54 = { add_core_of = FRC }
		55 = { add_core_of = FRC }
		59 = { add_core_of = FRC }
		959 = { add_core_of = FRC }
		64 = { add_core_of = FRC }
		63 = { add_core_of = FRC } # Provence.
		961 = { add_core_of = FRC }
		# Independence is not an autonomy tier in MD's subject-state definitions.
		release = FRC
		if = {
			limit = { FRC = { exists = yes } }
			set_country_flag = rev_fra_left_split_started
			remove_ideas = pluralist
			add_ideas = christian
			# Add reciprocal cores after release so the initial territorial split stays intact.
			every_owned_state = { add_core_of = FRC }
			77 = { # Piemonte.
				if = {
					limit = { is_owned_by = ITA }
					transfer_state_to = FRC
					add_core_of = FRC
				}
			}
			80 = { # Lombardia.
				if = {
					limit = { is_owned_by = ITA }
					transfer_state_to = FRC
					add_core_of = FRC
				}
			}
			1059 = { # Aosta Valley.
				if = {
					limit = { is_owned_by = ITA }
					transfer_state_to = FRC
					add_core_of = FRC
				}
			}
			1060 = { # Liguria.
				if = {
					limit = { is_owned_by = ITA }
					transfer_state_to = FRC
					add_core_of = FRC
				}
			}
			FRC = {
				every_owned_state = { add_core_of = FRA }
				if = {
					limit = { is_subject = yes }
					overlord = { set_autonomy = { target = FRC autonomy_state = autonomy_free } }
				}
				leave_faction = yes
				set_capital = { state = 56 }
				set_variable = { gdp_per_capita = FRA.gdp_per_capita }
				set_variable = { corporate_tax_rate = FRA.corporate_tax_rate }
				set_variable = { population_tax_rate = FRA.population_tax_rate }
				update_laws_for_new_nation = yes
				set_temp_variable = { nation_to_copy_from = FRA.id }
				copy_internal_factions = yes
				remove_ideas = christian
				add_ideas = pluralist
				rev_frc_setup_left_coalition = yes
				ingame_startup_nations_effect = yes
				# Use live research, rather than hardcoding the scenario's old tech list.
				inherit_technology = FRA
				set_research_slots = FRA.amount_research_slots
				if = {
					limit = { owns_state = 961 }
					if = {
						limit = { NOT = { has_tech = land_Drone_equipment3 } }
						set_technology = { land_Drone_equipment3 = 1 }
					}
					load_oob = FRC_split_drone_brigades
					if = {
						limit = { owns_state = 77 }
						load_oob = FRC_split_piemonte_drone_brigades
					}
					if = {
						limit = { owns_state = 63 }
						load_oob = FRC_split_provence_drone_brigades
					}
					if = {
						limit = { owns_state = 1060 }
						load_oob = FRC_split_liguria_drone_brigades
					}
				}
				if = {
					limit = { owns_state = 77 }
					load_oob = FRC_additional_italian_defectors
					load_oob = FRC_split_italian_reinforcements
				}
				if = {
					limit = { owns_state = 961 }
					if = {
						limit = { has_dlc = "No Step Back" }
						load_oob = FRC_split_spr_army_nsb
						load_oob = FRC_split_spanish_reinforcements_nsb
					}
					else = {
						load_oob = FRC_split_spr_army_legacy
						load_oob = FRC_split_spanish_reinforcements_legacy
					}
				}
				if = {
					limit = { has_dlc = "No Step Back" }
					load_oob = FRC_split_belgian_reinforcements_nsb
				}
				else = { load_oob = FRC_split_belgian_reinforcements_legacy }
				set_country_flag = rev_frc_left_split_established
			}
			# Removes the tagged original formations wherever they have moved, then
			# re-forms those same historical units under FRC. No random army fraction.
			rev_fra_reassign_regional_army = yes
			if = {
				limit = { has_country_flag = rev_fra_provence_formation_tagged FRC = { owns_state = 63 } }
				delete_unit = { id = 9905801 disband = no }
				FRC = { load_oob = FRC_split_army_legacy_provence }
			}
			if = {
				limit = { FRC = { owns_state = 63 } }
				endtimes_reassign_frc_french_reinforcements = yes
			}
			ITA = {
				rev_ita_reassign_frc_volunteers = yes
				if = {
					limit = { exists = yes FRC = { owns_state = 77 } }
					endtimes_reassign_additional_italian_defectors = yes
					endtimes_reassign_frc_italian_reinforcements = yes
				}
			}
			SPR = {
				rev_spr_reassign_frc_volunteers = yes
				if = {
					limit = { exists = yes FRC = { owns_state = 961 } }
					endtimes_reassign_frc_spanish_reinforcements = yes
				}
			}
			BEL = {
				if = {
					limit = { exists = yes }
					endtimes_reassign_frc_belgian_reinforcements = yes
				}
			}
			ingame_update_setup = yes
			if = {
				limit = { NOT = { has_global_flag = endtimes_french_civil_war_announced } }
				set_global_flag = endtimes_french_civil_war_announced
				news_event = { id = endtimes.22 }
			}
			SWI = { country_event = { id = endtimes.59 days = 1 } }
			ENG = { country_event = { id = endtimes.57 days = 2 } }
			# Leadership changes after five full days, independently of popup dismissal.
			country_event = { id = endtimes.11 days = 5 }
		}
		else = { log = "[GetDateText]: France could not release the French Popular Republic." }
	}
}
