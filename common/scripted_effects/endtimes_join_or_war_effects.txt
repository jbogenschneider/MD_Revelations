# Mark every affected player before sending any of the choices.
endtimes_offer_join_or_war = {
	set_global_flag = endtimes_join_or_war_offered
	every_country = {
		limit = { endtimes_join_or_war_participant = yes is_ai = no }
		set_country_flag = endtimes_join_or_war_response_pending
	}
	every_country = {
		limit = { has_country_flag = endtimes_join_or_war_response_pending }
		country_event = { id = endtimes.71 }
	}
	endtimes_try_apply_join_or_war = yes
}

endtimes_try_apply_join_or_war = {
	if = {
		limit = {
			has_global_flag = endtimes_join_or_war_offered
			NOT = { has_global_flag = endtimes_join_or_war_completed }
			NOT = {
				any_country = {
					exists = yes
					is_ai = no
					has_country_flag = endtimes_join_or_war_response_pending
				}
			}
		}
		endtimes_apply_join_or_war = yes
		endtimes_broadcast_news_46 = yes
	}
}

# Global scope. Prepare every surviving participant before enrolling any of them.
# Settle wars against the destination bloc or fellow entrants; other wars continue.
endtimes_apply_join_or_war = {
	set_global_flag = endtimes_join_or_war_completed
	every_country = {
		limit = { endtimes_join_or_war_participant = yes }
		save_event_target_as = endtimes_ultimatum_applicant
		if = {
			limit = {
				is_subject = yes
				NOT = {
					AND = { endtimes_join_or_war_saudi_subject = yes is_subject_of = SAU }
				}
			}
			overlord = {
				set_autonomy = { target = event_target:endtimes_ultimatum_applicant autonomy_state = autonomy_free }
			}
		}
		every_other_country = {
			limit = {
				OR = {
					tag = SAU
					rev_is_meccapact_faction = yes
					endtimes_join_or_war_participant = yes
				}
				NOT = { has_country_flag = endtimes_join_or_war_opted_out }
				has_war_with = event_target:endtimes_ultimatum_applicant
			}
			white_peace = event_target:endtimes_ultimatum_applicant
		}
		if = {
			limit = { NOT = { rev_is_meccapact_faction = yes } }
			MeccaPact_leave_previous_alliance = yes
		}
	}
	every_country = {
		limit = {
			endtimes_join_or_war_participant = yes
			NOT = { rev_is_nato_faction = yes }
			NOT = { rev_is_westnato_faction = yes }
		}
		if = {
			limit = {
				NOT = { has_government = communism }
				OR = {
					tag = SLO
					tag = CZE
					tag = HUN
					tag = SER
					tag = ALB
					tag = KOS
					endtimes_join_or_war_saudi_subject = yes
				}
			}
			set_country_leader_ideology = Autocracy
			if = {
				limit = { endtimes_join_or_war_saudi_subject = yes }
				set_politics = { ruling_party = communism elections_allowed = no }
			}
			else = { set_politics = { ruling_party = communism } }
			clear_array = ruling_party
			clear_array = gov_coalition_array
			add_to_array = { ruling_party = 7 } # Emerging Autocracy.
			clr_country_flag = party8_banned
			startup_politics = yes
		}
		if = {
			limit = {
				OR = {
					tag = BUL
					tag = AZE
					tag = ARM
					tag = GEO
					tag = MOR
					tag = TUN
					tag = AUS
				}
			}
			set_country_leader_ideology = Conservative
			set_politics = { ruling_party = communism elections_allowed = yes }
			clear_array = ruling_party
			clear_array = gov_coalition_array
			add_to_array = { ruling_party = 6 } # Emerging conservatism with elections.
			clr_country_flag = party7_banned
			startup_politics = yes
		}
		MeccaPact_join = yes
		if = {
			limit = {
				endtimes_join_or_war_saudi_subject = yes
				SAU = { exists = yes is_subject = no }
				NOT = { is_subject_of = SAU }
			}
			save_event_target_as = endtimes_ultimatum_saudi_subject
			SAU = {
				puppet = {
					target = event_target:endtimes_ultimatum_saudi_subject
					end_wars = no
					end_civil_wars = no
				}
			}
		}
		endtimes_apply_meccapact_ultimatum_benefits = yes
	}
}

endtimes_apply_meccapact_ultimatum_benefits = {
	if = {
		limit = { rev_is_meccapact_faction = yes }
		add_ideas = Solomon_one_world_currency
		add_ideas = endtimes_cabal
		add_ideas = endtimes_mark
		set_country_flag = rev_solomon_currency_adopted
		set_country_flag = endtimes_cabal_initialized
		endtimes_refresh_support_floor = yes
		ingame_update_setup = yes
	}
}
