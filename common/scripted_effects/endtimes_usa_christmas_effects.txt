# USA scope. Christmas Consecration preserves the elected government.
endtimes_restore_usa_at_christmas = {
	set_country_flag = endtimes_usa_christmas_reconciliation_completed
	set_country_flag = endtimes_usa_christian_restoration

	# Return to the dollar and prevent later re-adoption through the USA focuses.
	remove_ideas = endtimes_cabal
	remove_ideas = Solomon_one_world_currency
	remove_ideas = USA_soloman_idea
	remove_ideas = USA_solomoney_idea
	remove_ideas = USA_punish_dollar_avoidance_idea
	set_country_flag = endtimes_cabal_initialized
	set_country_flag = rev_solomon_currency_adopted

	# Re-release the western bloc using its established cores, including those
	# retained from the earlier resistance split. Only USA-owned land changes hands.
	if = { limit = { NOT = { country_exists = CAL } } release = CAL }
	every_state = {
		limit = { is_owned_by = USA is_core_of = CAL }
		transfer_state_to = CAL
	}
	CAL = {
		if = {
			limit = { exists = yes }
			if = {
				limit = { is_subject_of = USA }
				USA = { set_autonomy = { target = CAL autonomy_state = autonomy_free } }
			}
			load_focus_tree = { tree = REV_cal_focus keep_completed = yes }
			# A new fictional president replaces the old California leadership.
			retire_country_leader = yes
			set_politics = { ruling_party = democratic elections_allowed = yes }
			random_list = {
				1 = {
					create_country_leader = {
						name = "Adrian Mercer"
						ideology = liberalism
						expire = "2050.1.1"
						traits = { career_politician }
					}
				}
				1 = {
					create_country_leader = {
						name = "Daniel Whitmore"
						ideology = liberalism
						expire = "2050.1.1"
						traits = { career_politician }
					}
				}
				1 = {
					create_country_leader = {
						name = "Nathan Cole"
						ideology = liberalism
						expire = "2050.1.1"
						traits = { career_politician }
					}
				}
			}
			add_ideas = Solomon_one_world_currency
			add_ideas = endtimes_cabal
			set_country_flag = rev_solomon_currency_adopted
			set_country_flag = endtimes_cabal_initialized
			endtimes_refresh_support_floor = yes
			ingame_update_setup = yes
		}
	}
	ingame_update_setup = yes
	rev_grant_holy_monarchy = yes
}
