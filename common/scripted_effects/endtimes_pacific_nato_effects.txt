# Country scope. Use the normal NATO helper and clear any former CSTO records.
endtimes_pacific_join_nato = {
	if = {
		limit = { exists = yes is_subject = no }
		if = {
			limit = { has_idea = CSTO_member }
			remove_ideas = CSTO_member
			remove_from_array = { global.CSTO_member = THIS }
			remove_from_tech_sharing_group = CSTO_Tech_Share
		}
		endtimes_ensure_nato_membership = yes
	}
}

# November 16 event. Japan dissolves its own bloc before anyone joins NATO.
endtimes_pacific_nato_realignment = {
	if = {
		limit = { NOT = { has_global_flag = endtimes_pacific_nato_realignment_completed } }
		set_global_flag = endtimes_pacific_nato_realignment_completed
		JAP = {
			if = {
				limit = {
					exists = yes
					is_faction_leader = yes
					NOT = { rev_is_nato_faction = yes }
				}
				dismantle_faction = yes
			}
			endtimes_pacific_join_nato = yes
		}
		AST = { endtimes_pacific_join_nato = yes }
		RAJ = { endtimes_pacific_join_nato = yes }
		PHI = { endtimes_pacific_join_nato = yes }
		KOR = { endtimes_pacific_join_nato = yes }
		NZL = { endtimes_pacific_join_nato = yes }
	}
}
