# Discourage Russian AI naval landings in Ukraine, including Odesa.
# Uses the same invade penalty as vanilla's East Prussia invasion deterrent.
rev_SOV_avoid_ukraine_naval_invasions = {
	allowed = { original_tag = SOV }
	enable = {
		is_ai = yes
		has_war_with = UKR
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = invade
		id = UKR
		value = -2000
	}
}
