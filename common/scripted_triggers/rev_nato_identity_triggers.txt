# Recognize template NATO and the established alliance in older saves.
rev_is_nato_faction = {
	is_in_faction = yes
	NOT = { rev_is_westnato_faction = yes }
	NOT = { rev_is_meccapact_faction = yes }
	NOR = {
		has_faction_template = faction_template_csto
		has_faction_template = faction_template_holy_league
	}
	OR = {
		AND = {
			has_global_flag = rev_nato_lock_ger_leader
			GER = { exists = yes is_faction_leader = yes }
			OR = { tag = GER is_in_faction_with = GER }
		}
		AND = {
			NOT = { has_global_flag = rev_nato_lock_ger_leader }
			OR = {
				has_faction_template = faction_template_nato
				AND = {
					GER = { is_in_faction_with = USA }
					OR = { tag = GER is_in_faction_with = GER }
				}
			}
		}
	}
}

# The election repair must find an existing alliance outside the French splinter.
rev_is_fra_election_nato_destination = {
	is_faction_leader = yes
	NOT = { tag = FRA }
	NOR = {
		rev_is_westnato_faction = yes
		rev_is_meccapact_faction = yes
		has_faction_template = faction_template_csto
		has_faction_template = faction_template_holy_league
	}
	OR = {
		AND = {
			# The original NATO can be stored as a generic faction in older saves.
			NOT = { has_global_flag = rev_nato_lock_ger_leader }
			GER = { is_in_faction_with = USA }
			OR = { tag = GER is_in_faction_with = GER }
		}
		AND = {
			rev_is_nato_faction = yes
			OR = {
				tag = GER
				tag = USA
				is_in_faction_with = GER
				is_in_faction_with = USA
			}
		}
	}
}

rev_fra_election_nato_transfer_pending = {
	tag = FRA
	is_subject = no
	has_government = democratic
	has_country_flag = rev_fra_2027_handover_done
	has_country_flag = rev_fra_2027_le_pen_elected
	OR = {
		has_country_leader = { name = "Marine Le Pen" ruling_only = yes }
		AND = {
			has_country_flag = rev_fra_2027_le_pen_western_handover
			has_country_leader_ideology = conservatism
		}
	}
	OR = {
		NOT = { has_country_flag = rev_fra_2027_nato_transfer_completed }
		is_faction_leader = yes
	}
	NOR = {
		has_country_flag = rev_nato_anti_usa_opt_out_resolved_fra
		has_country_flag = endtimes_meccapact_realignment_completed
		has_war_with = USA
		has_idea = CSTO_member
		has_faction_template = faction_template_csto
		rev_is_westnato_faction = yes
		rev_is_meccapact_faction = yes
		AND = {
			has_global_flag = endtimes_meccapact_realignment_dispatched
			endtimes_meccapact_realignment_country = yes
		}
	}
	any_other_country = { rev_is_fra_election_nato_destination = yes }
}

# Match the saved split directly when the United States still leads the main faction.
rev_nato_has_us_french_split = {
	NOT = { has_global_flag = rev_nato_lock_ger_leader }
	USA = { exists = yes is_faction_leader = yes }
	FRA = {
		exists = yes
		is_faction_leader = yes
		NOT = { is_in_faction_with = USA }
	}
	FIN = { is_in_faction_with = FRA }
}

# FRA scope. Match the saved French splinter even if its template check fails.
rev_nato_is_french_splinter = {
	tag = FRA
	is_faction_leader = yes
	has_global_flag = rev_nato_canada_reorganized
	GER = { rev_is_nato_faction = yes }
	NOR = {
		is_in_faction_with = GER
		rev_is_westnato_faction = yes
		rev_is_meccapact_faction = yes
		has_faction_template = faction_template_csto
		has_faction_template = faction_template_holy_league
		any_other_country = { is_in_faction_with = FRA NOT = { tag = FIN } }
	}
	OR = {
		FIN = { is_in_faction_with = FRA }
		has_country_flag = rev_nato_french_splinter_repair_pending
	}
}

# Country scope. Preserve wars and subject relationships during save repair.
rev_nato_can_reunite_member = {
	is_subject = no
	NOT = { original_tag = ISR }
	NOT = {
		any_country = {
			OR = {
				tag = event_target:rev_nato_destination
				is_in_faction_with = event_target:rev_nato_destination
			}
			has_war_with = PREV
		}
	}
}
