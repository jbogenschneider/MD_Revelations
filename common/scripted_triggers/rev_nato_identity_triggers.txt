# Recognize template NATO and the established German-led faction in older saves.
# The fallback requires the existing storyline leadership lock, not just Germany.
rev_is_nato_faction = {
	is_in_faction = yes
	NOT = { rev_is_westnato_faction = yes }
	NOT = { has_faction_template = faction_template_meccapact }
	NOT = { has_faction_template = faction_template_csto }
	OR = {
		has_faction_template = faction_template_nato
		AND = {
			has_global_flag = rev_nato_lock_ger_leader
			GER = { exists = yes is_faction_leader = yes }
			OR = { tag = GER is_in_faction_with = GER }
		}
	}
}
