# The seal reports are triggered-only news; date triggers alone cannot fire them.
on_actions = {
	on_daily = {
		effect = {
			# December 25, 2028: NATO receives Angelic Protection.
			if = {
				limit = {
					date > 2028.12.24
					date < 2028.12.26
					NOT = { has_global_flag = endtimes_sealed_announced }
				}
				set_global_flag = endtimes_sealed_announced
				endtimes_broadcast_news_seal_7 = yes
			}
			# December 11, 2028: worldwide report on the Mark.
			if = {
				limit = {
					date > 2028.12.10
					date < 2028.12.12
					NOT = { has_global_flag = endtimes_mark_announced }
				}
				set_global_flag = endtimes_mark_announced
				endtimes_broadcast_news_seal_6 = yes
			}
			# October 2: announce and found the Saudi-led MeccaPact.
			if = {
				limit = {
					NOT = { date < 2028.10.2 }
					date < 2028.10.3
					SAU = { exists = yes }
					NOT = { has_global_flag = endtimes_alliance_expansion_announced }
				}
				endtimes_found_saudi_meccapact = yes
				if = {
					limit = { SAU = { rev_is_meccapact_faction = yes is_faction_leader = yes } }
					set_global_flag = endtimes_alliance_expansion_announced
					endtimes_broadcast_news_25 = yes
				}
			}
			if = {
				limit = {
					date > 2027.1.5
					date < 2027.1.7
					NOT = { has_global_flag = endtimes_first_seal_announced }
				}
				set_global_flag = endtimes_first_seal_announced
				endtimes_broadcast_news_seal_1 = yes
			}
			if = {
				limit = {
					date > 2027.8.5
					date < 2027.8.7
					NOT = { has_global_flag = endtimes_second_seal_announced }
				}
				set_global_flag = endtimes_second_seal_announced
				endtimes_broadcast_news_seal_2 = yes
			}
			if = {
				limit = {
					date > 2027.10.5
					date < 2027.10.7
					NOT = { has_global_flag = endtimes_third_seal_announced }
				}
				set_global_flag = endtimes_third_seal_announced
				endtimes_broadcast_news_seal_3 = yes
			}
			if = {
				limit = {
					date > 2027.12.5
					date < 2027.12.7
					NOT = { has_global_flag = endtimes_fourth_seal_announced }
				}
				set_global_flag = endtimes_fourth_seal_announced
				endtimes_broadcast_news_seal_4 = yes
			}
			if = {
				limit = {
					NOT = { date < 2029.1.6 }
					date < 2029.1.7
					NOT = { has_global_flag = endtimes_fifth_seal_announced }
				}
				set_global_flag = endtimes_fifth_seal_announced
				endtimes_broadcast_news_seal_5 = yes
			}
			if = {
				limit = {
					has_global_flag = endtimes_fifth_seal_announced
					NOT = { date < 2029.1.6 }
					date < 2029.4.4
					NOT = { check_variable = { global.endtimes_disaster_population_last_day = global.num_days } }
					# Capture the starting population, then apply losses at the monthly ticks and before the WNN report.
					OR = {
						NOT = { has_variable = global.endtimes_disaster_population_start_day }
						AND = { NOT = { date < 2029.2.1 } date < 2029.2.2 }
						AND = { NOT = { date < 2029.3.1 } date < 2029.3.2 }
						AND = { NOT = { date < 2029.4.1 } date < 2029.4.2 }
						NOT = { date < 2029.4.3 }
					}
				}
				set_variable = { global.endtimes_disaster_population_last_day = global.num_days }
				endtimes_update_disaster_population = yes
			}
			# Keep the spirits on active countries, including saves that missed the Fifth Seal.
			if = {
				limit = {
					NOT = { date < 2029.1.6 }
					exists = yes
				}
				if = {
					limit = { endtimes_uses_disasters_spirit = yes }
					if = {
						limit = { has_idea = endtimes_disaster_famine_and_disease }
						remove_ideas = endtimes_disaster_famine_and_disease
					}
					if = {
						limit = { NOT = { has_idea = endtimes_disasters } }
						add_ideas = endtimes_disasters
					}
				}
				else = {
					if = {
						limit = { has_idea = endtimes_disasters }
						remove_ideas = endtimes_disasters
					}
					if = {
						limit = { NOT = { has_idea = endtimes_disaster_famine_and_disease } }
						add_ideas = endtimes_disaster_famine_and_disease
					}
				}
			}
			# Broadcast after the last population update.
			if = {
				limit = {
					NOT = { date < 2029.4.3 }
					date < 2029.4.4
					NOT = { has_global_flag = endtimes_grim_mark_announced }
				}
				set_global_flag = endtimes_grim_mark_announced
				if = {
					limit = { SAU = { exists = yes } }
					SAU = { news_event = { id = endtimes.72 } }
				}
				else = { news_event = { id = endtimes.72 } }
			}
		}
	}
}
