# Global scheduling for the fictional Endtimes storyline.
on_actions = {
	on_startup = {
		effect = {
			# The 2026 bookmark starts at 08:00 on January 1; date triggers accept days only.
			if = {
				limit = {
					date > 2025.12.31
					date < 2026.1.2
				}
				# Membership applies to AI and human countries; only humans see the notices.
				every_country = {
					limit = {
						NOT = { has_country_flag = endtimes_cabal_initialized }
						OR = {
							tag = USA
							tag = SOV
							tag = CHI
							tag = UAE
							tag = SAU
							tag = RAJ
							tag = ARG
						}
					}
					set_country_flag = endtimes_cabal_initialized
					add_ideas = endtimes_cabal
				}
				every_country = {
					limit = {
						is_ai = no
						NOT = { has_country_flag = endtimes_intro_scheduled }
						OR = {
							tag = USA
							tag = SOV
							tag = CHI
							tag = UAE
							tag = SAU
							tag = RAJ
							tag = ARG
						}
					}
					set_country_flag = endtimes_intro_scheduled
					country_event = { id = endtimes.1 }
					# Schedule independently of the player's acknowledgement.
					if = {
						limit = { tag = USA }
						country_event = { id = endtimes.2 hours = 2 }
					}
				}
			}
		}
	}

	on_daily = {
		effect = {
			# Three days after the Global Ceasefire, restore constitutional USA.
			if = {
				limit = {
					tag = USA
					has_global_flag = { flag = endtimes_global_ceasefire_occurred days > 2 }
					NOT = { has_country_flag = endtimes_usa_returns_to_normal }
				}
				country_event = { id = endtimes.44 }
			}
			# Jeffries remains interim president until the November election.
			if = {
				limit = {
					tag = USA
					has_country_flag = endtimes_usa_returns_to_normal
					has_country_flag = rev_usa_cw_restoration_transition
					NOT = { has_country_flag = rev_usa_cw_restoration_kelly_done }
					date > 2028.11.9
				}
				country_event = { id = rev_usa_cw.43 }
			}
			# November 13, 2028: Iran dissolves its bloc and joins MeccaPact.
			if = {
				limit = {
					tag = PER
					date > 2028.11.12
					date < 2028.11.14
					is_subject = no
					SAU = { exists = yes is_subject = no }
					NOT = { has_country_flag = endtimes_iran_meccapact_completed }
				}
				country_event = { id = endtimes.26 }
			}
			# December 21, 2028: Christmas Consecration and California's separation.
			if = {
				limit = { endtimes_usa_christmas_reconciliation_ready = yes }
				country_event = { id = endtimes.41 }
			}
			# October 7, 2028: government choice on both prior paths.
			if = {
				limit = {
					tag = SOV
					date > 2028.10.6
					date < 2028.10.8
					NOT = { has_country_flag = endtimes_russia_october_coup_completed }
					NOT = { has_country_flag = endtimes_russia_october_elections_completed }
				}
				country_event = { id = endtimes.39 }
			}
			# Three days after the Miracle, apply peace and broadcast endtimes.38.
			if = {
				limit = {
					has_global_flag = endtimes_news_4_dispatched
					has_global_flag = { flag = endtimes_miracle_occurred days > 2 }
					NOT = { has_global_flag = endtimes_global_ceasefire_occurred }
				}
				endtimes_apply_global_ceasefire = yes
			}
			# August 7, 2026: a defense pact, without creating an in-game faction.
			if = {
				limit = {
					date > 2026.8.6
					date < 2026.8.8
					NOT = { has_global_flag = endtimes_regional_defense_pact_formed }
					SAU = { exists = yes }
					TUR = { exists = yes }
					PAK = { exists = yes }
				}
				endtimes_form_regional_defense_pact = yes
			}
			# November 3, 2028: Israel's coup also broadcasts the global WNN report.
			if = {
				limit = {
					tag = ISR
					date > 2028.11.2
					date < 2028.11.4
					SOV = { exists = yes }
					NOT = { has_country_flag = endtimes_israel_coup_completed }
				}
				country_event = { id = endtimes.28 }
			}
			# November 5, 2028: worldwide WNN report, broadcast once.
			if = {
				limit = {
					date > 2028.11.4
					date < 2028.11.6
					NOT = { has_global_flag = endtimes_moment_of_desolation_announced }
				}
				set_global_flag = endtimes_moment_of_desolation_announced
				endtimes_broadcast_news_30 = yes
			}
			# November 10, 2028: China joins Alex's alliance and attacks East Asia.
			if = {
				limit = {
					tag = CHI
					date > 2028.11.9
					date < 2028.11.11
					SOV = { exists = yes }
					NOT = { has_country_flag = endtimes_china_alliance_entry_completed }
				}
				country_event = { id = endtimes.31 }
			}
			# November 16, 2028: the Asian Quad joins NATO.
			if = {
				limit = {
					tag = AST
					date > 2028.11.15
					date < 2028.11.17
					NOT = { has_global_flag = endtimes_pacific_nato_realignment_completed }
				}
				country_event = { id = endtimes.33 }
			}
			# Israeli nuclear response; the follow-up ceasefire is queued one day later.
			if = {
				limit = {
					tag = ISR
					has_war_with = PER
					AND = {
						NOT = { controls_province = 1201 }
						NOT = { controls_province = 7107 }
						NOT = { controls_province = 1065 }
					}
					NOT = { has_country_flag = endtimes_israel_khuzestan_strike_done }
				}
				country_event = { id = endtimes.34 }
			}
			# Broadcast each worldwide WNN announcement once; global flags prevent duplicates.
			if = {
				limit = {
					date > 2027.11.23
					date < 2027.11.25
					NOT = { has_global_flag = endtimes_bad_harvest_occurred }
				}
				set_global_flag = endtimes_bad_harvest_occurred
				endtimes_broadcast_news_13 = yes
			}
			if = {
				limit = {
					date > 2028.1.4
					date < 2028.1.6
					NOT = { has_global_flag = endtimes_jerusalem_dirty_bomb_occurred }
					PER = {
						exists = yes
						controls_province = 16215 # Tehran
					}
				}
				set_global_flag = endtimes_jerusalem_dirty_bomb_occurred
				endtimes_broadcast_news_8 = yes
			}
			if = {
				limit = {
					tag = PER
					date > 2028.1.11
					date < 2028.1.13
					has_global_flag = endtimes_jerusalem_dirty_bomb_occurred
					NOT = { has_global_flag = endtimes_tehran_nuked }
					controls_province = 16215
					ISR = { exists = yes }
				}
				country_event = { id = endtimes.9 }
			}
			# January 13, 2028: worldwide report after the scripted Tehran strike.
			if = {
				limit = {
					date > 2028.1.12
					date < 2028.1.14
					has_global_flag = endtimes_tehran_nuked
					NOT = { has_global_flag = endtimes_tehran_strike_announced }
				}
				set_global_flag = endtimes_tehran_strike_announced
				endtimes_broadcast_news_24 = yes
			}
			if = {
				limit = {
					endtimes_great_warning_date = yes
					NOT = { has_global_flag = endtimes_great_warning_march_2028_dispatched }
				}
				# An older scheduler could consume the report in January. Reset the
				# timestamp here so the three-day follow-up starts from the real Warning.
				clr_global_flag = endtimes_great_warning_occurred
				set_global_flag = endtimes_great_warning_occurred
				set_global_flag = endtimes_great_warning_march_2028_dispatched
				set_global_flag = endtimes_news_3_dispatched
				endtimes_broadcast_news_3 = yes
			}
			if = {
				limit = {
					date > 2028.4.12
					date < 2028.4.14
					NOT = { has_global_flag = endtimes_news_4_dispatched }
				}
				set_global_flag = endtimes_miracle_occurred
				set_global_flag = endtimes_news_4_dispatched
				endtimes_broadcast_news_4 = yes
			}
			if = {
				limit = {
					date > 2028.5.5
					date < 2028.5.7
					has_global_flag = endtimes_miracle_occurred
					NOT = { has_global_flag = endtimes_news_19_dispatched }
				}
				set_global_flag = endtimes_orthodox_reunion_occurred
				set_global_flag = endtimes_news_19_dispatched
				endtimes_broadcast_news_19 = yes
			}
			if = {
				limit = {
					date > 2028.6.5
					date < 2028.6.7
					NOT = { has_global_flag = endtimes_news_5_dispatched }
				}
				set_global_flag = endtimes_new_church_emerged
				set_global_flag = endtimes_news_5_dispatched
				endtimes_broadcast_news_5 = yes
			}
			if = {
				limit = {
					endtimes_prophets_announce_date = yes
					NOT = { has_global_flag = endtimes_news_6_dispatched }
				}
				set_global_flag = endtimes_prophets_announce_occurred
				set_global_flag = endtimes_news_6_dispatched
				endtimes_broadcast_news_6 = yes
			}
			if = {
				limit = {
					endtimes_prophets_miracle_date = yes
					NOT = { has_global_flag = endtimes_news_666_dispatched }
				}
				set_global_flag = endtimes_prophets_miracle_occurred
				set_global_flag = endtimes_news_666_dispatched
				endtimes_broadcast_news_666 = yes
			}
			if = {
				limit = {
					date > 2028.11.6
					date < 2028.11.7
					NOT = { has_global_flag = endtimes_news_20_dispatched }
				}
				set_global_flag = endtimes_one_faith_announced
				set_global_flag = endtimes_news_20_dispatched
				endtimes_broadcast_news_20 = yes
			}
			if = {
				limit = {
					date > 2028.10.1
				date < 2028.10.3
					SAU = { exists = yes is_subject = no }
					NOT = { has_global_flag = endtimes_meccapact_realignment_dispatched }
				}
				set_global_flag = endtimes_meccapact_realignment_dispatched
				# Settle wars and leave old alliances before any country joins MeccaPact.
				every_country = {
					limit = { endtimes_meccapact_realignment_country = yes }
					endtimes_prepare_meccapact_realignment = yes
				}
				SAU = { MeccaPact_join = yes set_faction_leader = yes }
				every_country = {
					limit = {
						endtimes_meccapact_realignment_country = yes
						NOT = { has_country_flag = endtimes_meccapact_realignment_completed }
					}
					country_event = { id = endtimes.16 }
				}
			}
			if = {
				limit = {
					tag = HLS
					date > 2028.10.30
					date < 2028.10.31
					NOT = { has_country_flag = endtimes_false_prophet_installed }
				}
				country_event = { id = endtimes.7 }
			}
			if = {
				limit = {
					date > 2028.11.19
					date < 2028.11.21
					NOT = { has_global_flag = endtimes_prophecy_fulfilled }
				}
				set_global_flag = endtimes_prophecy_fulfilled
				endtimes_broadcast_news_12 = yes
			}
			if = {
				limit = {
					date > 2028.11.16
					date < 2028.11.19
					NOT = { has_global_flag = endtimes_europe_realigned }
				}
				set_global_flag = endtimes_europe_realigned
				endtimes_broadcast_news_10 = yes
			}
			if = {
				limit = {
					date > 2029.6.4
					date < 2029.6.6
					NOT = { has_global_flag = endtimes_famine_disasters_announced }
				}
				set_global_flag = endtimes_famine_disasters_announced
				endtimes_broadcast_news_21 = yes
			}
			if = {
				limit = {
					tag = FRA
					date > 2028.11.21
					date < 2028.11.23
					NOT = { has_country_flag = rev_fra_left_split_started }
					NOT = { country_exists = FRC }
					has_variable = rev_fra_regional_oob_mode
					owns_state = 56
					owns_state = 54	
					owns_state = 961
					owns_state = 958
					owns_state = 61
					owns_state = 951
					owns_state = 959
					owns_state = 60
				}
				# The former French monarchy date now starts the civil-war chain.
				country_event = { id = rev_fra_split.1 }
			}
			if = {
				limit = {
					tag = GER
					date > 2028.11.26
					date < 2028.11.28
					NOT = { has_country_flag = endtimes_german_monarchy_restored }
				}
				country_event = { id = endtimes.14 }
			}
			if = {
				limit = {
					has_global_flag = endtimes_famine_disasters_announced
					NOT = { has_country_flag = endtimes_famine_spirit_initialized }
					OR = {
						tag = MAD
						tag = SUD
						tag = ETH
						tag = SOM
					}
				}
				set_country_flag = endtimes_famine_spirit_initialized
				add_ideas = endtimes_famine_and_disease
			}
			else_if = {
				limit = { has_idea = endtimes_famine_and_disease }
				endtimes_refresh_famine_consumption_floor = yes
			}
			if = {
				limit = { has_idea = endtimes_cabal }
				endtimes_refresh_support_floor = yes
			}
		}
	}
}
