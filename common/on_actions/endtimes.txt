# Global scheduling for the fictional Endtimes storyline.
on_actions = {
	on_startup = {
		effect = {
			# The 2026 bookmark starts at 08:00 on January 1; date triggers accept days only.
			if = {
				limit = {
					date > 2025.12.31
					date < 2026.1.2
				}
				# Membership applies to AI and human countries; only humans see the notices.
				every_country = {
					limit = {
						NOT = { has_country_flag = endtimes_cabal_initialized }
						OR = {
							tag = USA
							tag = SOV
							tag = CHI
							tag = UAE
							tag = SAU
							tag = RAJ
							tag = ARG
						}
					}
					set_country_flag = endtimes_cabal_initialized
					add_ideas = endtimes_cabal
				}
				every_country = {
					limit = {
						is_ai = no
						NOT = { has_country_flag = endtimes_intro_scheduled }
						OR = {
							tag = USA
							tag = SOV
							tag = CHI
							tag = UAE
							tag = SAU
							tag = RAJ
							tag = ARG
						}
					}
					set_country_flag = endtimes_intro_scheduled
					country_event = { id = endtimes.1 }
					# Schedule independently of the player's acknowledgement.
					if = {
						limit = { tag = USA }
						country_event = { id = endtimes.2 hours = 2 }
					}
				}
			}
		}
	}

	on_daily = {
		effect = {
			# August 7, 2026: a defense pact, without creating an in-game faction.
			if = {
				limit = {
					date > 2026.8.6
					date < 2026.8.8
					NOT = { has_global_flag = endtimes_regional_defense_pact_formed }
					SAU = { exists = yes }
					TUR = { exists = yes }
					PAK = { exists = yes }
				}
				endtimes_form_regional_defense_pact = yes
			}
			# November 3, 2028: Israel's coup also broadcasts the global WNN report.
			if = {
				limit = {
					tag = ISR
					date > 2028.11.2
					date < 2028.11.4
					SOV = { exists = yes }
					NOT = { has_country_flag = endtimes_israel_coup_completed }
				}
				country_event = { id = endtimes.28 }
			}
			# November 5, 2028: worldwide WNN report, broadcast once.
			if = {
				limit = {
					date > 2028.11.4
					date < 2028.11.6
					NOT = { has_global_flag = endtimes_moment_of_desolation_announced }
				}
				set_global_flag = endtimes_moment_of_desolation_announced
				news_event = { id = endtimes.30 }
			}
			# November 10, 2028: China joins Alex's alliance and attacks East Asia.
			if = {
				limit = {
					tag = CHI
					date > 2028.11.9
					date < 2028.11.11
					SOV = { exists = yes }
					NOT = { has_country_flag = endtimes_china_alliance_entry_completed }
				}
				country_event = { id = endtimes.31 }
			}
			# November 11, 2028: Australia's wartime alignment choice.
			if = {
				limit = {
					tag = AST
					date > 2028.11.10
					date < 2028.11.12
					NOT = { has_country_flag = endtimes_australia_choice_presented }
				}
				country_event = { id = endtimes.33 }
			}
			# Israeli nuclear response; the follow-up ceasefire is queued one day later.
			if = {
				limit = {
					tag = ISR
					has_war_with = PER
					OR = {
						NOT = { controls_province = 1201 }
						NOT = { controls_province = 7107 }
					}
					NOT = { has_country_flag = endtimes_israel_khuzestan_strike_done }
				}
				country_event = { id = endtimes.34 }
			}
			# Broadcast each worldwide WNN announcement once; global flags prevent duplicates.
			if = {
				limit = {
					date > 2027.11.23
					date < 2027.11.25
					NOT = { has_global_flag = endtimes_bad_harvest_occurred }
				}
				set_global_flag = endtimes_bad_harvest_occurred
				news_event = { id = endtimes.13 }
			}
			if = {
				limit = {
					date > 2028.1.4
					date < 2028.1.6
					NOT = { has_global_flag = endtimes_jerusalem_dirty_bomb_occurred }
					PER = {
						exists = yes
						controls_province = 16215 # Tehran
					}
				}
				set_global_flag = endtimes_jerusalem_dirty_bomb_occurred
				news_event = { id = endtimes.8 }
			}
			if = {
				limit = {
					tag = SOV
					date > 2028.1.5
					date < 2028.1.7
					NOT = { has_country_flag = endtimes_alex_took_over }
				}
				country_event = { id = endtimes.15 }
			}
			if = {
				limit = {
					tag = PER
					date > 2028.1.11
					date < 2028.1.13
					has_global_flag = endtimes_jerusalem_dirty_bomb_occurred
					NOT = { has_global_flag = endtimes_tehran_nuked }
					controls_province = 16215
					ISR = { exists = yes }
				}
				country_event = { id = endtimes.9 }
			}
			# January 13, 2028: worldwide report after the scripted Tehran strike.
			if = {
				limit = {
					date > 2028.1.12
					date < 2028.1.14
					has_global_flag = endtimes_tehran_nuked
					NOT = { has_global_flag = endtimes_tehran_strike_announced }
				}
				set_global_flag = endtimes_tehran_strike_announced
				news_event = { id = endtimes.24 }
			}
			if = {
				limit = {
					date > 2028.1.19
					date < 2028.1.21
					NOT = { has_global_flag = endtimes_great_warning_occurred }
				}
				set_global_flag = endtimes_great_warning_occurred
				news_event = { id = endtimes.3 }
			}
			if = {
				limit = {
					date > 2028.4.12
					date < 2028.4.14
					NOT = { has_global_flag = endtimes_miracle_occurred }
				}
				set_global_flag = endtimes_miracle_occurred
				news_event = { id = endtimes.4 }
			}
			if = {
				limit = {
					date > 2028.4.15
					date < 2028.4.17
					has_global_flag = endtimes_miracle_occurred
					NOT = { has_global_flag = endtimes_orthodox_reunion_occurred }
				}
				set_global_flag = endtimes_orthodox_reunion_occurred
				news_event = { id = endtimes.19 }
			}
			if = {
				limit = {
					date > 2028.6.5
					date < 2028.6.7
					NOT = { has_global_flag = endtimes_new_church_emerged }
				}
				set_global_flag = endtimes_new_church_emerged
				news_event = { id = endtimes.5 }
			}
			if = {
				limit = {
					date > 2029.3.29
					date < 2029.3.31
					NOT = { has_global_flag = endtimes_prophets_miracle_occurred }
				}
				set_global_flag = endtimes_prophets_miracle_occurred
				news_event = { id = endtimes.6 }
			}
			if = {
				limit = {
					date > 2029.4.28
					date < 2029.4.30
					NOT = { has_global_flag = endtimes_one_faith_announced }
				}
				set_global_flag = endtimes_one_faith_announced
				news_event = { id = endtimes.20 }
			}
			if = {
				limit = {
					date > 2028.10.1
				date < 2028.10.3
					SOV = { exists = yes }
					NOT = { has_global_flag = endtimes_csto_realignment_dispatched }
				}
				set_global_flag = endtimes_csto_realignment_dispatched
				# All participants leave their previous alliances before the realignment events.
				every_country = {
					limit = { endtimes_csto_realignment_country = yes }
					endtimes_leave_previous_alliance = yes
				}
				every_country = {
					limit = {
						endtimes_csto_realignment_country = yes
						NOT = { has_country_flag = endtimes_csto_realignment_completed }
					}
					country_event = { id = endtimes.16 }
				}
			}
			if = {
				limit = {
					tag = HLS
					date > 2029.5.1
					date < 2029.5.3
					NOT = { has_country_flag = endtimes_false_prophet_installed }
				}
				country_event = { id = endtimes.7 }
			}
			if = {
				limit = {
					date > 2029.5.9
					date < 2029.5.11
					NOT = { has_global_flag = endtimes_prophecy_fulfilled }
				}
				set_global_flag = endtimes_prophecy_fulfilled
				news_event = { id = endtimes.12 }
			}
			if = {
				limit = {
					date > 2029.6.1
					date < 2029.6.3
					NOT = { has_global_flag = endtimes_europe_realigned }
				}
				set_global_flag = endtimes_europe_realigned
				news_event = { id = endtimes.10 }
			}
			if = {
				limit = {
					date > 2029.6.4
					date < 2029.6.6
					NOT = { has_global_flag = endtimes_famine_disasters_announced }
				}
				set_global_flag = endtimes_famine_disasters_announced
				news_event = { id = endtimes.21 }
			}
			if = {
				limit = {
					tag = FRA
					date > 2029.7.1
					date < 2029.7.3
					NOT = { has_country_flag = rev_fra_left_split_started }
					NOT = { country_exists = FRC }
					has_variable = rev_fra_regional_oob_mode
					owns_state = 56
					owns_state = 54
					owns_state = 55
					owns_state = 305
					owns_state = 59
				}
				# The former French monarchy date now starts the civil-war chain.
				country_event = { id = rev_fra_split.1 }
			}
			if = {
				limit = {
					tag = GER
					date > 2029.7.2
					date < 2029.7.4
					NOT = { has_country_flag = endtimes_german_monarchy_restored }
				}
				country_event = { id = endtimes.14 }
			}
			if = {
				limit = {
					has_global_flag = endtimes_famine_disasters_announced
					NOT = { has_country_flag = endtimes_famine_spirit_initialized }
					OR = {
						tag = MAD
						tag = SUD
						tag = ETH
						tag = SOM
					}
				}
				set_country_flag = endtimes_famine_spirit_initialized
				add_ideas = endtimes_famine_and_disease
			}
			else_if = {
				limit = { has_idea = endtimes_famine_and_disease }
				endtimes_refresh_famine_consumption_floor = yes
			}
			if = {
				limit = { has_idea = endtimes_cabal }
				endtimes_refresh_support_floor = yes
			}
		}
	}
}
